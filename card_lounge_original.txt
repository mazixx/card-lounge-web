def tl_combo(cards):
    if not cards or len(set(cards)) != len(cards):
        return None

    ordered = sorted(cards, key=tl_key)
    size = len(ordered)
    ranks = [tl_key(card)[0] for card in ordered]
    high = tl_key(ordered[-1])
    counts = Counter(ranks)

    if size == 1:
        return ("single", 1, high)

    if size in (2, 3, 4) and len(counts) == 1:
        return ("same", size, high)

    # Sảnh không chứa 2.
    if (
        size >= 3
        and 15 not in counts
        and len(counts) == size
        and max(counts) - min(counts) == size - 1
    ):
        return ("straight", size, high)

    # Chỉ coi 3 hoặc 4 đôi thông là bộ dùng để đánh/chặt.
    # Có 5+ đôi thông thì chọn một đoạn 3 hoặc 4 đôi.
    if size in (6, 8):
        pair_count = size // 2
        values = sorted(counts)

        if (
            15 not in counts
            and len(values) == pair_count
            and all(count == 2 for count in counts.values())
            and values[-1] - values[0] == pair_count - 1
        ):
            return ("pairs", size, high)

    return None


def tl_beats(cards, table):
    new = tl_combo(cards)

    if new is None:
        return False

    if not table:
        return True

    old = tl_combo(table)

    if old is None:
        return False

    # Đè cùng loại, cùng số lá.
    if new[:2] == old[:2]:
        return new[2] > old[2]

    single_two = len(table) == 1 and table[0][0] == 2

    pair_two = (
        len(table) == 2
        and old[:2] == ("same", 2)
        and all(card[0] == 2 for card in table)
    )

    old_three_pairs = old[:2] == ("pairs", 6)
    old_four_kind = old[:2] == ("same", 4)

    # Ba đôi thông chặt một lá 2.
    if new[:2] == ("pairs", 6):
        return single_two

    # Tứ quý chặt 2 lẻ hoặc ba đôi thông.
    if new[:2] == ("same", 4):
        return single_two or old_three_pairs

    # Bốn đôi thông chặt các bộ đặc biệt.
    if new[:2] == ("pairs", 8):
        return (
            single_two
            or pair_two
            or old_three_pairs
            or old_four_kind
        )

    return False


def tl_candidates(hand):
    hand = sorted(hand, key=tl_key)
    output = [(card,) for card in hand]
    groups = {}

    for card in hand:
        groups.setdefault(tl_key(card)[0], []).append(card)

    # Đôi, bộ ba, tứ quý.
    for group in groups.values():
        for size in (2, 3, 4):
            output.extend(itertools.combinations(group, size))

    # Sảnh: giữ lá thấp ở các bậc dưới;
    # xét mọi chất ở lá cuối để so độ lớn.
    for low in range(3, 13):
        for high in range(low + 2, 15):
            if all(rank in groups for rank in range(low, high + 1)):
                base = [
                    groups[rank][0]
                    for rank in range(low, high)
                ]

                for last in groups[high]:
                    output.append(tuple(base + [last]))

    # Ba và bốn đôi thông.
    for pair_count in (3, 4):
        for low in range(3, 16 - pair_count):
            values = list(range(low, low + pair_count))

            if not all(
                rank in groups and len(groups[rank]) >= 2
                for rank in values
            ):
                continue

            choices = [
                list(itertools.combinations(groups[rank], 2))
                for rank in values
            ]

            for selected_pairs in itertools.product(*choices):
                output.append(
                    tuple(
                        card
                        for pair in selected_pairs
                        for card in pair
                    )
                )

    return output

@lru_cache(maxsize=24000)
def tl_plan(hand):
    """
    Ước lượng cách chia bài thành các lượt đánh.

    Trả về:
        (số bộ cần đánh, số lá phải đánh lẻ)

    Ưu tiên:
    1. Ít lượt đánh.
    2. Ít lá rác.

    Đây là tìm kiếm trên các bộ do tl_candidates sinh ra,
    không phải tìm kiếm toàn bộ diễn biến với đối thủ.
    """
    hand = tuple(sorted(hand, key=tl_key))

    if not hand:
        return (0, 0)

    anchor = hand[0]
    best = (99, 99)
    visited = set()

    # Chọn bộ chứa lá đầu để tránh xét nhiều thứ tự
    # khác nhau của cùng một cách chia bài.
    for combo in tl_candidates(hand):
        if anchor not in combo:
            continue

        chosen = frozenset(combo)

        if chosen in visited:
            continue

        visited.add(chosen)

        remaining = tuple(
            card for card in hand
            if card not in chosen
        )

        turns, singles = tl_plan(remaining)

        candidate = (
            turns + 1,
            singles + (1 if len(combo) == 1 else 0),
        )

        if candidate < best:
            best = candidate

    return best

@lru_cache(maxsize=24000)
def solve_phom(hand, protected=()):
    hand = tuple(hand)
    required = 0
    protected = set(protected)

    for i, c in enumerate(hand):
        if c in protected:
            required |= 1 << i

    melds = set()

    for rank in range(2, 15):
        indices = [i for i, c in enumerate(hand) if c[0] == rank]
        for n in (3, 4):
            for combo in itertools.combinations(indices, n):
                melds.add(sum(1 << i for i in combo))

    for suit in range(4):
        lookup = {
            phom_value(c): i
            for i, c in enumerate(hand)
            if c[1] == suit
        }

        for low in range(1, 12):
            for high in range(low + 2, 14):
                if all(v in lookup for v in range(low, high + 1)):
                    melds.add(
                        sum(1 << lookup[v] for v in range(low, high + 1))
                    )

    containing = [[] for _ in hand]

    for mask in melds:
        for i in range(len(hand)):
            if mask & (1 << i):
                containing[i].append(mask)

    @lru_cache(None)
    def search(mask):
        if mask == 0:
            return 0, ()

        bit = mask & -mask
        i = bit.bit_length() - 1
        best = (9999, ())

        if not required & bit:
            score, groups = search(mask ^ bit)
            best = (score + phom_value(hand[i]), groups)

        for meld in containing[i]:
            if mask & meld == meld:
                score, groups = search(mask ^ meld)
                candidate = (score, (meld,) + groups)

                if candidate[0] < best[0]:
                    best = candidate

        return best

    score, masks = search((1 << len(hand)) - 1)

    groups = tuple(
        tuple(hand[i] for i in range(len(hand)) if mask & (1 << i))
        for mask in masks
    )
    return score, groups


def phom_result(hand, eaten):
    """
    Tìm cách chia phỏm có tổng điểm bài rác thấp nhất.

    Luật:
    - Phỏm cùng số: 3 hoặc 4 lá.
    - Phỏm sảnh: ít nhất 3 lá liên tiếp cùng chất.
    - A đứng trước 2, không nối Q-K-A.
    - Mỗi phỏm có tối đa MỘT lá đã ăn.
    - Mọi lá đã ăn bắt buộc nằm trong một phỏm.
    - Một lá không được dùng trong hai phỏm.

    Trả về:
        (điểm bài rác, danh sách phỏm)

    Không có cách ghép hợp lệ:
        (9999, [])
    """
    from itertools import combinations
    from functools import lru_cache

    cards = tuple(hand)
    eaten_set = set(eaten)

    if (
        len(cards) != len(set(cards))
        or not eaten_set.issubset(set(cards))
    ):
        return 9999, []

    count = len(cards)

    def rank(card):
        return 1 if card[0] == 14 else card[0]

    def is_meld(group):
        # Mỗi phỏm chỉ có tối đa một lá ăn.
        if sum(card in eaten_set for card in group) > 1:
            return False

        # Bộ ba hoặc bộ bốn cùng số.
        if len({card[0] for card in group}) == 1:
            return len(group) in (3, 4)

        # Sảnh cùng chất, ít nhất ba lá liên tiếp.
        if len({card[1] for card in group}) != 1:
            return False

        values = sorted(rank(card) for card in group)

        return (
            len(set(values)) == len(group)
            and values[-1] - values[0] == len(group) - 1
        )

    # Mỗi phỏm hợp lệ được biểu diễn bằng bitmask.
    melds_by_card = [[] for _ in cards]

    for size in range(3, count + 1):
        for indexes in combinations(range(count), size):
            group = [cards[index] for index in indexes]

            if not is_meld(group):
                continue

            mask = sum(1 << index for index in indexes)

            for index in indexes:
                melds_by_card[index].append(mask)

    @lru_cache(maxsize=None)
    def solve(mask):
        if mask == 0:
            return 0, ()

        first = (mask & -mask).bit_length() - 1
        first_bit = 1 << first

        best_score = 9999
        best_groups = ()

        # Lá chưa ăn có thể để làm bài rác.
        # Lá đã ăn không được phép bỏ ngoài phỏm.
        if cards[first] not in eaten_set:
            tail_score, tail_groups = solve(mask ^ first_bit)

            if tail_score < 9999:
                best_score = (
                    phom_value(cards[first]) + tail_score
                )
                best_groups = tail_groups

        # Thử mọi phỏm chứa lá đầu tiên còn lại.
        for meld_mask in melds_by_card[first]:
            if mask & meld_mask != meld_mask:
                continue

            tail_score, tail_groups = solve(mask ^ meld_mask)

            if tail_score >= 9999:
                continue

            candidate_groups = (meld_mask,) + tail_groups

            if (
                tail_score < best_score
                or (
                    tail_score == best_score
                    and len(candidate_groups) > len(best_groups)
                )
            ):
                best_score = tail_score
                best_groups = candidate_groups

        return best_score, best_groups

    score, group_masks = solve((1 << count) - 1)

    if score >= 9999:
        return 9999, []

    groups = []

    for mask in group_masks:
        group = [
            cards[index]
            for index in range(count)
            if mask & (1 << index)
        ]

        group.sort(
            key=lambda card: (rank(card), card[1])
        )

        groups.append(group)

    return score, groups

def test_phom_one_eaten():
    # Bốn lá Q, chỉ một lá đã ăn: hợp lệ.
    hand = [(12, 0), (12, 1), (12, 2), (12, 3)]

    score, groups = phom_result(hand, [(12, 2)])
    assert score == 0
    assert len(groups) == 1

    # Cùng bộ bốn Q nhưng có hai lá đã ăn: không hợp lệ.
    score, groups = phom_result(
        hand,
        [(12, 2), (12, 3)],
    )
    assert score == 9999

    # Hai phỏm riêng, mỗi phỏm một lá ăn: hợp lệ.
    hand = [
        (12, 0), (12, 1), (12, 2),
        (7, 3), (8, 3), (9, 3),
    ]

    score, groups = phom_result(
        hand,
        [(12, 2), (9, 3)],
    )
    assert score == 0
    assert len(groups) == 2

    print("Đã kiểm tra: mỗi phỏm tối đa một lá ăn.")


test_phom_one_eaten()

def poker_five(cards):
    ranks = sorted((c[0] for c in cards), reverse=True)
    counts = Counter(ranks)
    groups = sorted(
        ((count, rank) for rank, count in counts.items()),
        reverse=True,
    )

    flush = len({c[1] for c in cards}) == 1
    unique = sorted(set(ranks), reverse=True)
    straight = 0

    if len(unique) == 5:
        if unique[0] - unique[-1] == 4:
            straight = unique[0]
        elif unique == [14, 5, 4, 3, 2]:
            straight = 5

    if flush and straight:
        return (8, straight)

    if groups[0][0] == 4:
        return (7, groups[0][1], groups[1][1])

    if groups[0][0] == 3 and groups[1][0] == 2:
        return (6, groups[0][1], groups[1][1])

    if flush:
        return (5, *ranks)

    if straight:
        return (4, straight)

    if groups[0][0] == 3:
        kickers = sorted(
            (r for r, count in counts.items() if count == 1),
            reverse=True,
        )
        return (3, groups[0][1], *kickers)

    pairs = sorted(
        (r for r, count in counts.items() if count == 2),
        reverse=True,
    )

    if len(pairs) == 2:
        kicker = next(r for r, count in counts.items() if count == 1)
        return (2, *pairs, kicker)

    if pairs:
        kickers = sorted(
            (r for r, count in counts.items() if count == 1),
            reverse=True,
        )
        return (1, pairs[0], *kickers)

    return (0, *ranks)


def poker_best(cards):
    return max(
        poker_five(combo)
        for combo in itertools.combinations(cards, 5)
    )


def equity_estimate(hole, board, opponents, trials=70):
    """
    Không đọc bài đối thủ.
    Giả định đối thủ cầm bài ngẫu nhiên từ các lá chưa biết.
    Kết quả là phần pot kỳ vọng, có tính trường hợp hòa.
    """
    if opponents <= 0:
        return 1.0

    known = set(hole + board)
    pool = [c for c in deck52() if c not in known]
    missing = 5 - len(board)
    credit = 0.0

    for _ in range(trials):
        sample = random.sample(pool, missing + 2 * opponents)
        community = board + sample[:missing]
        own = poker_best(hole + community)

        others = [
            poker_best(
                sample[missing + 2*j:missing + 2*j + 2] + community
            )
            for j in range(opponents)
        ]

        best = max(others)

        if own > best:
            credit += 1
        elif own == best:
            credit += 1 / (1 + sum(value == own for value in others))

    return credit / trials

def cao_score(hand):
    if all(c[0] in (11, 12, 13) for c in hand):
        return (1, 0)

    total = sum(
        1 if r == 14 else (0 if r >= 10 else r)
        for r, _ in hand
    )
    return (0, total % 10)


def cao_label(hand):
    score = cao_score(hand)
    return "Ba Tây" if score[0] else f"{score[1]} nút"

class Engine:

    def say(self, text):
        self.logger(text)
    def transfer_tournament_points(
        self, loser_seat, winner_seat, amount, reason
    ):
        """Chuyển điểm giải ngay sau một hành động hợp lệ."""
        tournament = getattr(self, "score_tournament", None)

        # Ván nhanh hoặc trò khác: không chuyển điểm.
        if (
            tournament is None
            or self.kind not in ("Tiến lên", "Tá lả")
        ):
            return

        if tournament.kind != self.kind:
            raise RuntimeError("Sai liên kết giữa ván và giải.")

        if (
            not isinstance(loser_seat, int)
            or not isinstance(winner_seat, int)
            or not 0 <= loser_seat < self.n
            or not 0 <= winner_seat < self.n
            or loser_seat == winner_seat
        ):
            raise RuntimeError("Ghế chuyển điểm không hợp lệ.")

        if amount <= 0:
            raise ValueError("Số điểm chuyển phải lớn hơn 0.")

        loser_pid = self.ids[loser_seat]
        winner_pid = self.ids[winner_seat]

        # Ghế thay đổi giữa các ván, nên luôn đổi sang ID.
        for pid in (loser_pid, winner_pid):
            if (
                pid not in tournament.points
                or pid not in tournament.total_points
            ):
                raise RuntimeError(
                    "Người chơi không thuộc bảng điểm hiện tại."
                )

        # Điểm dùng xét đi tiếp ở bàn hiện tại.
        # Không giới hạn về 0: được phép âm.
        tournament.points[loser_pid] -= amount
        tournament.points[winner_pid] += amount

        # Điểm tích lũy toàn giải.
        tournament.total_points[loser_pid] -= amount
        tournament.total_points[winner_pid] += amount

        if not hasattr(self, "point_transfers"):
            self.point_transfers = []

        self.point_transfers.append({
            "loser_pid": loser_pid,
            "winner_pid": winner_pid,
            "amount": amount,
            "reason": reason,
        })

        self.say(
            f"💸 {reason}: "
            f"{self.name(loser_seat)} -{amount:g} điểm → "
            f"{self.name(winner_seat)} +{amount:g} điểm."
        )
    def __init__(self, kind, ids, hand_number=0, logger=None):
        self.kind = kind
        self.ids = list(ids)
        self.n = len(ids)
        self.logger = logger or (lambda text: None)

        self.deck = deck52()
        random.shuffle(self.deck)

        self.hands = [[] for _ in ids]
        self.chips = [1000] * self.n
        self.bets = [0] * self.n

        self.board = []
        self.table = []
        self.turn = -1
        self.done = False

        # Mỗi phần tử là một nhóm đồng hạng.
        self.ranking = []
        self.status = ""
        self.folded = set()
        self.button = hand_number % self.n

        if kind == "Tiến lên":
            self.start_tl()
        elif kind == "Tá lả":
            self.start_phom()
        elif kind == "Poker":
            self.start_poker()
        else:
            self.start_cao()

    def name(self, seat):
        return player_name(self.ids[seat])

    def say(self, text):
        self.logger(text)

    def deal(self, count):
        for _ in range(count):
            for hand in self.hands:
                hand.append(self.deck.pop())

    def finish(self, groups):
        self.ranking = [list(group) for group in groups if group]
        self.done = True
        self.turn = -1

        winners = ", ".join(self.name(i) for i in self.ranking[0])
        self.status = f"🏆 Nhất ván: {winners}"
        self.say(self.status)

    def groups_by(self, values, reverse=False):
        ordered = sorted(range(self.n), key=lambda i: values[i], reverse=reverse)
        return [
            list(group)
            for _, group in itertools.groupby(ordered, key=lambda i: values[i])
        ]

    def human_seat(self):
        return self.ids.index(0) if 0 in self.ids else None

    def visible(self, seat):
        if self.ids[seat] == 0:
            return True

        if self.kind == "Poker":
            return (
                self.done
                and getattr(self, "showdown", False)
                and seat not in self.folded
            )

        if self.kind == "Ba cào":
            return seat in self.revealed or self.done

        return self.done

    def options(self):
        if self.done:
            return []

        if self.kind == "Tiến lên":
            if self.tl_reacting:
                return [
                    ("⚡ Chặt ngoài lượt", "cut"),
                    ("Bỏ qua rình", "skip_cut"),
                ]

            return [
                ("Đánh bài", "play"),
                ("Bỏ lượt", "pass"),
            ]

        if self.kind == "Tá lả":
            if self.phase == "draw":
                return [("Bốc bài", "draw"), ("Ăn bài", "eat")]
            return [("Đánh lá đã chọn", "discard")]

        if self.kind == "Poker":
            return [
                ("Bỏ bài", "fold"),
                ("Check / Theo", "call"),
                (f"Tố +{self.limit}", "raise"),
            ]

        return [("Lật bài / So bài", "reveal")]

    def action(self, action, selected=()):
        if self.done:
            return

        if self.kind == "Tiến lên":
            self.action_tl(action, selected)
        elif self.kind == "Tá lả":
            self.action_phom(action, selected)
        elif self.kind == "Poker":
            self.action_poker(action)
        else:
            self.action_cao(action)

    def bot(self):
        if self.done:
            return

        i = self.turn

        if self.kind == "Tiến lên":
            cards = self.tl_recommend(i)

            if self.tl_reacting:
                if cards:
                    self.action("cut", cards)
                else:
                    self.action("skip_cut")
            else:
                if cards:
                    self.action("play", cards)
                else:
                    self.action("pass")

        elif self.kind == "Tá lả":
            if self.phase == "draw":
                self.action("eat" if self.can_eat(i) else "draw")
            else:
                choices = self.phom_discards(i)
                self.action("discard", [choices[0][2]])

        elif self.kind == "Poker":
            strength = self.poker_strength(i)
            need = self.current - self.bets[i]
            odds = need / (sum(self.contrib) + need) if need else 0

            if need and strength < odds + 0.04 and random.random() < 0.88:
                self.action("fold")
            elif (
                self.raise_count < 3
                and (strength > 0.61 or random.random() < 0.045)
            ):
                self.action("raise")
            else:
                self.action("call")
        else:
            self.action("reveal")

    def start_tl(self):
        self.deal(13)

        self.hands = [
            sorted(hand, key=tl_key)
            for hand in self.hands
        ]

        self.first = True
        self.finished = []
        self.thoi_two = []
        self.passed = set()
        self.tl_seen = set()

        self.tl_reacting = False
        self.tl_reaction_queue = []

        # Hạng thực của từng ghế; chỉ được điền khi kết thúc.
        self.rank_map = {}
        self.white_labels = {}

        self.turn = next(
            seat
            for seat, hand in enumerate(self.hands)
            if (3, 0) in hand
        )
        self.leader = self.turn

        if self.tl_check_white():
            return

        self.status = (
            f"{self.name(self.turn)} mở ván, phải chứa 3♠."
        )

        self.say(
            "Tiến lên: có đôi thông, rình bằng 4 đôi thông; "
            "không về bằng 2; cùng thối cùng hạng 4."
        )

    def tl_white_type(self, hand):
        """
        Số ưu tiên nhỏ hơn là mạnh hơn.
        Không có tứ quý 3 hoặc ba đôi thông chứa 3♠.
        """
        counts = Counter(card[0] for card in hand)

        if len(hand) == 13 and len(counts) == 13:
            return (0, "Sảnh rồng")

        # Tứ quý được tính thành hai đôi.
        pair_count = sum(count // 2 for count in counts.values())

        if pair_count >= 6:
            return (1, "Sáu đôi")

        consecutive = 0
        longest = 0

        # 3..A, không chứa 2.
        for rank in range(3, 15):
            if counts.get(rank, 0) >= 2:
                consecutive += 1
                longest = max(longest, consecutive)
            else:
                consecutive = 0

        if longest >= 5:
            return (2, "Năm đôi thông trở lên")

        if counts.get(2, 0) == 4:
            return (3, "Tứ quý 2")

        return None

    def tl_check_white(self):
        results = {
            seat: self.tl_white_type(hand)
            for seat, hand in enumerate(self.hands)
        }

        qualified = {
            seat: result
            for seat, result in results.items()
            if result is not None
        }

        if not qualified:
            return False

        best_priority = min(
            result[0] for result in qualified.values()
        )

        winners = [
            seat
            for seat, result in qualified.items()
            if result[0] == best_priority
        ]

        self.white_labels = {
            seat: result[1]
            for seat, result in qualified.items()
        }

        for seat, result in qualified.items():
            self.say(
                f"✨ {self.name(seat)} có {result[1]}: "
                f"{cards_text(self.hands[seat])}"
            )

        ranks = {
            seat: (1 if seat in winners else self.n)
            for seat in range(self.n)
        }

        winning_label = qualified[winners[0]][1]

        self.tl_finish_ranks(
            ranks,
            "✨ TỚI TRẮNG — "
            + ", ".join(self.name(seat) for seat in winners)
            + f" ({winning_label})",
        )
        return True

    def tl_finish_ranks(self, ranks, status=None):
        self.rank_map = dict(ranks)
        self.tl_reacting = False
        self.tl_reaction_queue = []

        groups = [
            [
                seat
                for seat in range(self.n)
                if ranks[seat] == rank
            ]
            for rank in sorted(set(ranks.values()))
        ]

        self.finish(groups)

        if status:
            self.status = status
            self.say(status)

        self.say(
            "Xếp hạng: "
            + " | ".join(
                f"{self.name(seat)}: hạng {ranks[seat]}"
                for seat in sorted(
                    range(self.n),
                    key=lambda seat: (ranks[seat], seat),
                )
            )
        )

    def tl_active(self):
        return [
            seat
            for seat in range(self.n)
            if seat not in self.finished
            and seat not in self.thoi_two
        ]

    def next_active(self, seat):
        active = set(self.tl_active())

        for step in range(1, self.n + 1):
            other = (seat + step) % self.n

            if other in active:
                return other

        return -1

    def tl_invalid_finish(self, seat, cards):
        return (
            len(cards) == len(self.hands[seat])
            and any(card[0] == 2 for card in cards)
        )

    def tl_remaining_after(self, seat, cards):
        chosen = set(cards)

        return tuple(
            sorted(
                (
                    card
                    for card in self.hands[seat]
                    if card not in chosen
                ),
                key=tl_key,
            )
        )

    def tl_leaves_only_twos(self, seat, cards):
        remaining = self.tl_remaining_after(seat, cards)

        return bool(remaining) and all(
            card[0] == 2 for card in remaining
        )

    def tl_reaction_moves(self, seat):
        """
        Rình chỉ dùng đúng 4 đôi thông.
        Được rình dù đã bỏ lượt.
        """
        if (
            not self.table
            or seat == self.leader
            or seat not in self.tl_active()
        ):
            return []

        return [
            list(combo)
            for combo in tl_candidates(self.hands[seat])
            if tl_combo(combo)[:2] == ("pairs", 8)
            and tl_beats(combo, self.table)
            and not self.tl_invalid_finish(seat, combo)
        ]

    def tl_legal(self, seat):
        if seat not in self.tl_active():
            return []

        if self.tl_reacting:
            if seat != self.turn:
                return []
            return self.tl_reaction_moves(seat)

        return [
            list(combo)
            for combo in tl_candidates(self.hands[seat])
            if tl_beats(combo, self.table)
            and (not self.first or (3, 0) in combo)
            and not self.tl_invalid_finish(seat, combo)
        ]

    def tl_check_thoi(self, seat):
        if seat not in self.tl_active():
            return False

        hand = self.hands[seat]

        if hand and all(card[0] == 2 for card in hand):
            self.thoi_two.append(seat)
            self.passed.discard(seat)

            self.say(
                f"⚠ {self.name(seat)} THỐI 2: "
                f"{cards_text(hand)} — nhận hạng {self.n}."
            )

            return True

        return False

    def tl_finish_if_needed(self):
        active = self.tl_active()

        if len(active) > 1:
            return False

        ranks = {}

        for position, seat in enumerate(self.finished, 1):
            ranks[seat] = position

        if active:
            ranks[active[0]] = len(self.finished) + 1

        # Mọi người thối đều cùng hạng thấp nhất.
        for seat in self.thoi_two:
            ranks[seat] = self.n

        self.tl_finish_ranks(ranks)
        return True

    def tl_continue_normal(self, previous_seat):
        self.tl_reacting = False
        self.tl_reaction_queue = []

        active = self.tl_active()

        challengers = [
            seat
            for seat in active
            if seat != self.leader
            and seat not in self.passed
        ]

        if not challengers:
            old_leader = self.leader
            self.table = []
            self.passed.clear()

            self.turn = (
                old_leader
                if old_leader in active
                else self.next_active(old_leader)
            )

            self.status = (
                f"{self.name(self.turn)} mở dây mới."
            )
            return

        for step in range(1, self.n + 1):
            seat = (previous_seat + step) % self.n

            if seat in challengers:
                self.turn = seat
                break

        self.status = f"Đến lượt {self.name(self.turn)}."

    def tl_open_reactions(self):
        """
        Xét theo chiều lượt từ người vừa đánh.
        Engine kiểm tra quyền hợp lệ; mỗi bot chỉ ra quyết định
        bằng bài của chính nó và thông tin công khai.
        """
        queue = []

        for step in range(1, self.n):
            seat = (self.leader + step) % self.n

            if self.tl_reaction_moves(seat):
                queue.append(seat)

        if not queue:
            self.tl_continue_normal(self.leader)
            return

        self.tl_reacting = True
        self.tl_reaction_queue = queue
        self.turn = queue[0]

        self.status = (
            f"⚡ {self.name(self.turn)} có quyền RÌNH "
            "bằng 4 đôi thông."
        )

    def tl_skip_reaction(self):
        seat = self.turn

        self.say(
            f"{self.name(seat)} bỏ qua cơ hội rình."
        )

        self.tl_reaction_queue.pop(0)

        if self.tl_reaction_queue:
            self.turn = self.tl_reaction_queue[0]

            self.status = (
                f"⚡ {self.name(self.turn)} có quyền RÌNH "
                "bằng 4 đôi thông."
            )
        else:
            self.tl_continue_normal(self.leader)

    def tl_apply_play(self, seat, cards, reaction=False):
        old_table = list(self.table)
        old_leader = self.leader
        new_combo = tl_combo(cards)
        old_combo = tl_combo(old_table)

        # Bộ hàng theo đúng luật hiện tại của file.
        special_types = {
            ("same", 4),   # Tứ quý
            ("pairs", 6),  # Ba đôi thông
            ("pairs", 8),  # Bốn đôi thông
        }

        cutting = False

        if (
            old_combo is not None
            and new_combo is not None
            and tl_beats(cards, old_table)
        ):
            new_type = new_combo[:2]
            old_type = old_combo[:2]

            cutting = (
                new_type in special_types
                and (
                    # Chặt khác loại: ví dụ tứ quý chặt 2.
                    new_type != old_type

                    # Chặt hàng cùng loại:
                    # tứ quý lớn chặt tứ quý nhỏ, v.v.
                    or old_type in special_types
                )
            )

        for card in cards:
            self.hands[seat].remove(card)

        self.table = sorted(cards, key=tl_key)
        self.leader = seat
        self.first = False
        self.tl_seen.update(cards)

        if reaction:
            # Rình thành công: người đã bỏ lượt được trở lại
            # với tư cách người đang giữ bộ trên bàn.
            self.passed.discard(seat)

        self.tl_reacting = False
        self.tl_reaction_queue = []

        if reaction:
            self.say(
                f"⚡ {self.name(seat)} RÌNH bằng "
                f"{cards_text(self.table)}, chặt "
                f"{cards_text(old_table)} "
                f"của {self.name(old_leader)}!"
            )
        elif cutting:
            self.say(
                f"💥 {self.name(seat)} dùng "
                f"{cards_text(self.table)} CHẶT "
                f"{cards_text(old_table)} "
                f"của {self.name(old_leader)}!"
            )
        else:
            self.say(
                f"{self.name(seat)} đánh "
                f"{cards_text(self.table)}."
            )

        # Bộ mới đã được đánh thành công.
        # Chuyển điểm trước khi xử lý về nhất/kết thúc ván.
        if cutting:
            two_count = sum(
                card[0] == 2
                for card in old_table
            )

            amount = (
                two_count * 0.5
                if two_count > 0
                else 0.5
            )

            self.transfer_tournament_points(
                loser_seat=old_leader,
                winner_seat=seat,
                amount=amount,
                reason=(
                    "Tiến lên chặt "
                    + cards_text(old_table)
                ),
            )

        if not self.hands[seat]:
            self.finished.append(seat)

            self.say(
                f"🏁 {self.name(seat)} về hợp lệ "
                f"hạng {len(self.finished)}."
            )
        else:
            self.tl_check_thoi(seat)

        if self.tl_finish_if_needed():
            return

        self.tl_open_reactions()

    def action_tl(self, action, selected):
        seat = self.turn

        if seat not in self.tl_active():
            raise ValueError(
                "Ghế này đã kết thúc ván."
            )

        if self.tl_reacting:
            if action == "skip_cut":
                self.tl_skip_reaction()
                return

            if action != "cut":
                raise ValueError(
                    "Đang ở cửa sổ rình. "
                    "Chọn Chặt ngoài lượt hoặc Bỏ qua."
                )

            cards = list(selected)

            if (
                len(cards) != 8
                or len(set(cards)) != 8
                or not all(
                    card in self.hands[seat]
                    for card in cards
                )
            ):
                raise ValueError(
                    "Hãy chọn đúng 8 lá tạo thành 4 đôi thông."
                )

            combo = tl_combo(cards)

            if (
                combo is None
                or combo[:2] != ("pairs", 8)
                or not tl_beats(cards, self.table)
            ):
                raise ValueError(
                    "Bốn đôi thông này chưa chặt được bộ trên bàn."
                )

            self.tl_apply_play(seat, cards, reaction=True)
            return

        if action == "pass":
            if not self.table:
                raise ValueError(
                    "Đang mở dây mới, bạn phải đánh bài."
                )

            self.passed.add(seat)
            self.say(f"{self.name(seat)} bỏ lượt.")
            self.tl_continue_normal(seat)
            return

        if action != "play":
            raise ValueError(
                "Hành động Tiến lên không hợp lệ."
            )

        cards = list(selected)

        if (
            not cards
            or len(set(cards)) != len(cards)
            or not all(
                card in self.hands[seat]
                for card in cards
            )
        ):
            raise ValueError(
                "Hãy chọn bài hợp lệ trên tay."
            )

        if self.tl_invalid_finish(seat, cards):
            raise ValueError(
                "Không được về bằng 2, "
                "kể cả đôi 2 hoặc bộ chỉ gồm các lá 2."
            )

        if not tl_beats(cards, self.table):
            raise ValueError(
                "Bộ bài không hợp lệ hoặc chưa đè/chặt được bài trên bàn."
            )

        if self.first and (3, 0) not in cards:
            raise ValueError(
                "Bộ mở ván phải chứa 3♠."
            )

        self.tl_apply_play(seat, cards)

    def tl_unseen_cards(self, seat):
        known = (
            set(self.hands[seat])
            | self.tl_seen
            | set(self.table)
        )

        return [
            card for card in deck52()
            if card not in known
        ]

    def tl_card_control(self, card, unseen):
        if not unseen:
            return 1.0

        higher = sum(
            tl_key(other) > tl_key(card)
            for other in unseen
        )

        return 1.0 - higher / len(unseen)

    def tl_dangerous_opponents(self, seat):
        return [
            other
            for other in self.tl_active()
            if other != seat
        ]

    def tl_move_score(self, seat, cards, unseen, opponents):
        remaining = self.tl_remaining_after(seat, cards)

        if not remaining:
            return 1_000_000.0

        if all(card[0] == 2 for card in remaining):
            return -1_000_000.0

        turns, singles = tl_plan(remaining)
        combo = tl_combo(cards)
        high_card = max(cards, key=tl_key)

        counts = [
            len(self.hands[other])
            for other in opponents
        ]

        nearest = min(counts, default=13)
        one_card_threat = 1 in counts

        score = (
            -110 * turns
            -23 * singles
            +8 * len(cards)
            -1.7 * tl_key(high_card)[0]
        )

        if len(cards) == 1 and nearest <= 2:
            score += 55 * self.tl_card_control(
                high_card, unseen
            )

        if len(remaining) == 1:
            score += 40
            score += 35 * self.tl_card_control(
                remaining[0], unseen
            )

        if len(remaining) <= 5:
            score -= 20 * sum(
                card[0] == 2 for card in remaining
            )

        if (
            len(self.hands[seat]) <= 5
            and any(card[0] == 2 for card in cards)
        ):
            score += 25

        if not self.table:
            if one_card_threat and len(cards) > 1:
                score += 160

            if 2 in counts:
                if len(cards) >= 3:
                    score += 75
                elif len(cards) == 2:
                    score -= 25
        else:
            cutting = combo[:2] != tl_combo(self.table)[:2]

            if cutting:
                score += 30

                if nearest <= 2:
                    score += 70

                if len(remaining) <= 3:
                    score += 35

        # Không tiêu hàng phòng thủ quá sớm nếu không cần.
        special = (
            combo[:2] == ("same", 4)
            or combo[0] == "pairs"
        )

        if (
            special
            and not self.table
            and len(remaining) > 4
            and nearest > 2
        ):
            score -= 30

        return score

    def tl_recommend(self, seat):
        choices = self.tl_legal(seat)

        if not choices:
            return []

        hand = self.hands[seat]

        finishing = [
            cards
            for cards in choices
            if len(cards) == len(hand)
        ]

        if finishing:
            return min(
                finishing,
                key=lambda cards: tl_combo(cards)[2],
            )

        safe = [
            cards
            for cards in choices
            if not self.tl_leaves_only_twos(seat, cards)
        ]

        if not safe:
            if self.table:
                return []

            return max(
                choices,
                key=lambda cards: (
                    len(cards),
                    tl_combo(cards)[2],
                ),
            )

        choices = safe
        opponents = self.tl_dangerous_opponents(seat)
        unseen = self.tl_unseen_cards(seat)

        # Rình: ưu tiên đường giải phóng bài tốt nhất.
        if self.tl_reacting:
            return max(
                choices,
                key=lambda cards: self.tl_move_score(
                    seat, cards, unseen, opponents
                ),
            )

        live_one_card_threat = any(
            len(self.hands[other]) == 1
            and (
                not self.table
                or other not in self.passed
            )
            for other in opponents
        )

        if live_one_card_threat:
            singles = [
                cards for cards in choices
                if len(cards) == 1
            ]

            multiple = [
                cards for cards in choices
                if len(cards) > 1
            ]

            if not self.table:
                if multiple:
                    choices = multiple
                elif singles:
                    return max(
                        singles,
                        key=lambda cards: tl_key(cards[0]),
                    )

            elif len(self.table) == 1:
                if self.table[0][0] == 2 and multiple:
                    choices = multiple
                elif singles:
                    return max(
                        singles,
                        key=lambda cards: tl_key(cards[0]),
                    )

        best = max(
            choices,
            key=lambda cards: (
                self.tl_move_score(
                    seat, cards, unseen, opponents
                ),
                len(cards),
                tl_combo(cards)[2],
            ),
        )

        if self.table and opponents:
            danger = any(
                len(self.hands[other]) <= 2
                for other in opponents
            )

            remaining = self.tl_remaining_after(seat, best)

            before_turns, before_singles = tl_plan(
                tuple(sorted(hand, key=tl_key))
            )
            after_turns, after_singles = tl_plan(remaining)

            extra_turns = 1 + after_turns - before_turns

            breaking_structure = (
                extra_turns >= 2
                or (
                    extra_turns >= 1
                    and after_singles > before_singles
                )
            )

            if (
                not danger
                and len(remaining) > 2
                and breaking_structure
            ):
                return []

        return best

    def start_phom(self):
        self.deal(9)
        self.hands[0].append(self.deck.pop())

        self.eaten = [[] for _ in self.ids]
        self.discard_count = [0] * self.n

        # Lưu bài đã công khai để đánh giá cạ chờ.
        self.phom_seen = set()

        # Các vùng hiển thị.
        # hands vẫn giữ TOÀN BỘ bài sở hữu để tương thích
        # với bot, tính điểm và kiểm tra ù hiện tại.
        self.phom_melds = [[] for _ in self.ids]
        self.phom_loose = [[] for _ in self.ids]
        self.phom_waiting = [[] for _ in self.ids]
        self.phom_trash = [[] for _ in self.ids]

        self.top = None
        self.top_owner = None
        self.table = []
        self.phase = "discard"
        self.turn = 0

        for seat in range(self.n):
            self.phom_refresh(seat)

        self.status = (
            f"{self.name(0)} có 10 lá, chọn một lá để đánh."
        )

        self.check_u(0)

    def phom_rank(self, card):
        """
        Giá trị dùng để xét sảnh/cạ.
        A đứng trước 2; không có Q-K-A.
        Không dùng điểm bài rác để xác định số liên tiếp.
        """
        rank = card[0]
        return 1 if rank == 14 else rank

    def phom_card_key(self, card):
        return (self.phom_rank(card), card[1])

    def phom_sort_meld(self, group):
        """Bộ cùng số xếp theo chất; sảnh xếp theo số."""
        group = list(group)

        if len({card[0] for card in group}) == 1:
            return sorted(group, key=lambda card: card[1])

        return sorted(group, key=self.phom_card_key)

    def phom_pair_waits(self, first, second):
        """
        Các lá có thể hoàn thành một cạ hai lá.

        Chỉ xét:
        - Hai lá cùng số.
        - Hai lá liền số cùng chất.
        - Hai lá cách một số cùng chất.
        """
        if first[0] == second[0]:
            return {
                (first[0], suit)
                for suit in range(4)
                if suit not in (first[1], second[1])
            }

        if first[1] != second[1]:
            return set()

        low, high = sorted((
            self.phom_rank(first),
            self.phom_rank(second),
        ))

        suit = first[1]
        waiting_ranks = []

        if high - low == 1:
            if low > 1:
                waiting_ranks.append(low - 1)

            if high < 13:
                waiting_ranks.append(high + 1)

        elif high - low == 2:
            waiting_ranks.append(low + 1)

        return {
            (14 if rank == 1 else rank, suit)
            for rank in waiting_ranks
        }

    def phom_arrange_loose(self, seat, cards):
        """
        Chọn các cạ không chồng lá.

        Ưu tiên cạ còn nhiều lá chờ chưa biết vị trí.
        Không đọc bài úp đối thủ hoặc nội dung nọc.
        """
        from functools import lru_cache

        cards = tuple(sorted(cards, key=self.phom_card_key))

        # Những lá này không được coi là lá chờ còn trong nọc.
        unavailable = (
            set(self.hands[seat])
            | self.phom_seen
        )

        pair_scores = {}

        for first in range(len(cards)):
            for second in range(first + 1, len(cards)):
                waits = self.phom_pair_waits(
                    cards[first],
                    cards[second],
                )

                available = waits - unavailable

                if not available:
                    continue

                same_rank = (
                    cards[first][0] == cards[second][0]
                )

                distance = abs(
                    self.phom_rank(cards[first])
                    - self.phom_rank(cards[second])
                )

                # Số lá chờ là yếu tố chính.
                # Khi bằng nhau: ưu tiên đôi, rồi cạ liền.
                preference = (
                    6 if same_rank
                    else 4 if distance == 1
                    else 2
                )

                pair_scores[(first, second)] = (
                    len(available) * 100 + preference
                )

        @lru_cache(maxsize=None)
        def choose(mask):
            if mask == 0:
                return (0, ())

            first = (mask & -mask).bit_length() - 1
            remaining = mask & ~(1 << first)

            # Có thể để lá đầu thành bài rác.
            best_score, best_pairs = choose(remaining)

            for second in range(first + 1, len(cards)):
                if not remaining & (1 << second):
                    continue

                score = pair_scores.get((first, second))

                if score is None:
                    continue

                tail_score, tail_pairs = choose(
                    remaining & ~(1 << second)
                )

                candidate = score + tail_score

                if candidate > best_score:
                    best_score = candidate
                    best_pairs = (
                        ((first, second),) + tail_pairs
                    )

            return best_score, best_pairs

        _, indexes = choose((1 << len(cards)) - 1)

        indexes = sorted(
            indexes,
            key=lambda pair: (
                -pair_scores[pair],
                self.phom_card_key(cards[pair[0]]),
            ),
        )

        pairs = [
            [cards[first], cards[second]]
            for first, second in indexes
        ]

        used = {
            card
            for pair in pairs
            for card in pair
        }

        trash = sorted(
            (
                card
                for card in cards
                if card not in used
            ),
            key=lambda card: (
                phom_value(card),
                self.phom_card_key(card),
            ),
        )

        ordered = [
            card
            for pair in pairs
            for card in pair
        ] + trash

        return ordered, pairs, trash

    def phom_refresh(self, seat):
        """
        Đồng bộ phỏm, bài rác và thứ tự bài.

        Không xóa bài phỏm khỏi hands.
        Hàm phom_result hiện có quyết định cách ghép hợp lệ,
        đồng thời bảo vệ mọi lá đã ăn.
        """
        score, groups = phom_result(
            self.hands[seat],
            self.eaten[seat],
        )

        if score >= 9999:
            raise RuntimeError(
                f"Bài của {self.name(seat)} không còn "
                "cách ghép hợp lệ bảo vệ lá đã ăn."
            )

        melds = [
            self.phom_sort_meld(group)
            for group in groups
        ]

        melds.sort(
            key=lambda group: (
                self.phom_card_key(group[0]),
                len(group),
            )
        )

        meld_cards = [
            card
            for group in melds
            for card in group
        ]

        # Chặn việc nhóm trả về bị trùng hoặc chứa bài lạ.
        if (
            len(meld_cards) != len(set(meld_cards))
            or not set(meld_cards).issubset(
                set(self.hands[seat])
            )
            or not set(self.eaten[seat]).issubset(
                set(meld_cards)
            )
        ):
            raise RuntimeError(
                "phom_result trả về các phỏm không hợp lệ."
            )

        grouped = set(meld_cards)

        loose = [
            card
            for card in self.hands[seat]
            if card not in grouped
        ]

        ordered, pairs, trash = self.phom_arrange_loose(
            seat, loose
        )

        self.phom_melds[seat] = melds
        self.phom_loose[seat] = ordered
        self.phom_waiting[seat] = pairs
        self.phom_trash[seat] = trash

        # Giao diện cũ vẫn vẽ được toàn bộ bài trong lúc
        # chưa cập nhật phần tách vùng phỏm.
        self.hands[seat] = ordered + meld_cards

    def phom_card_visible(self, seat, card):
        """
        Dùng khi vẽ lá trong khu vực phỏm.

        - Chủ bài thấy toàn bộ.
        - Lá đã ăn luôn công khai.
        - Kết thúc ván lật toàn bộ để tổng kết.
        """
        return (
            self.done
            or self.ids[seat] == 0
            or card in self.eaten[seat]
        )

    def can_eat(self, seat):
        if self.top is None:
            return False

        score, _ = phom_result(
            self.hands[seat] + [self.top],
            self.eaten[seat] + [self.top],
        )

        return score < 9999

    def phom_discards(self, seat):
        out = []

        # Phỏm đang đặt xuống không được chọn đánh trực tiếp.
        # Bài được đánh nằm trong vùng bài rác/cạ chờ.
        loose = set(self.phom_loose[seat])

        for card in self.hands[seat]:
            if card not in loose:
                continue

            if card in self.eaten[seat]:
                continue

            remainder = list(self.hands[seat])
            remainder.remove(card)

            score, groups = phom_result(
                remainder,
                self.eaten[seat],
            )

            if score < 9999:
                out.append((
                    score,
                    -phom_value(card),
                    card,
                    groups,
                ))

        return sorted(out)

    def phom_rankings(self):
        values = []

        for seat, hand in enumerate(self.hands):
            score, groups = phom_result(
                hand,
                self.eaten[seat],
            )

            values.append((not bool(groups), score))

        return self.groups_by(values)

    def check_u(self, seat):
        score, groups = phom_result(
            self.hands[seat],
            self.eaten[seat],
        )

        if score == 0 and groups:
            self.say(
                f"Ù! {self.name(seat)} không còn bài rác."
            )

            self.finish(self.phom_rankings())
            return True

        return False

    def action_phom(self, action, selected):
        seat = self.turn

        if self.phase == "draw":
            # Chỉ có giá trị nếu lần này thực sự ăn bài.
            eaten_from = None

            if action == "eat":
                if not self.can_eat(seat):
                    raise ValueError(
                        "Không thể ăn lá này.\n\n"
                        "• Lá ăn phải nằm trong một phỏm hợp lệ.\n"
                        "• Mỗi phỏm chỉ được chứa một lá đã ăn.\n"
                        "• Các lá đã ăn trước đó vẫn phải được "
                        "bảo vệ trong phỏm."
                    )

                card = self.top
                eaten_from = self.top_owner

                # Phải biết chính xác ai đã đánh lá này.
                if (
                    not isinstance(eaten_from, int)
                    or not 0 <= eaten_from < self.n
                    or eaten_from == seat
                ):
                    raise RuntimeError(
                        "Không xác định được chủ lá bài bị ăn."
                    )

                self.hands[seat].append(card)
                self.eaten[seat].append(card)
                self.phom_seen.add(card)

                self.say(
                    f"{self.name(seat)} ăn {card_text(card)}. "
                    "Lá đã ăn được công khai trong khu phỏm."
                )

            elif action == "draw":
                if not self.deck:
                    self.finish(self.phom_rankings())
                    return

                card = self.deck.pop()
                self.hands[seat].append(card)

                if self.ids[seat] == 0:
                    self.say(
                        f"Bạn bốc được {card_text(card)}."
                    )
                else:
                    self.say(
                        f"{self.name(seat)} bốc một lá."
                    )

            else:
                raise ValueError(
                    "Bạn cần ăn bài hoặc bốc bài trước."
                )

            self.top = None
            self.top_owner = None
            self.table = []

            # Hoàn tất ghép phỏm trước khi chuyển điểm.
            self.phom_refresh(seat)

            self.phase = "discard"
            self.status = (
                f"{self.name(seat)} chọn một lá bài rác để đánh."
            )

            if eaten_from is not None:
                self.transfer_tournament_points(
                    loser_seat=eaten_from,
                    winner_seat=seat,
                    amount=0.5,
                    reason=f"Tá lả ăn {card_text(card)}",
                )

            # Nếu ăn xong ù thì vẫn được nhận điểm ăn,
            # sau đó mới kết thúc ván.
            self.check_u(seat)
            return

        if action != "discard" or len(selected) != 1:
            raise ValueError(
                "Hãy chọn đúng một lá rồi bấm Đánh."
            )

        card = list(selected)[0]

        legal = {
            entry[2]
            for entry in self.phom_discards(seat)
        }

        if card not in legal:
            raise ValueError(
                "Chỉ được đánh bài trong vùng bài rác/cạ chờ. "
                "Không đánh lá đã nằm trong phỏm."
            )

        self.hands[seat].remove(card)
        self.discard_count[seat] += 1

        self.top = card
        self.top_owner = seat
        self.table = [card]
        self.phom_seen.add(card)

        self.say(
            f"{self.name(seat)} đánh {card_text(card)}."
        )

        # Bài vừa công khai có thể làm thay đổi chất lượng
        # cạ chờ của những người khác.
        for other in range(self.n):
            self.phom_refresh(other)

        if self.check_u(seat):
            return

        if all(
            count >= 4
            for count in self.discard_count
        ):
            for other, hand in enumerate(self.hands):
                score, groups = phom_result(
                    hand,
                    self.eaten[other],
                )

                text = (
                    " | ".join(
                        cards_text(group)
                        for group in groups
                    )
                    or "MÓM"
                )

                self.say(
                    f"{self.name(other)}: "
                    f"{score} điểm rác; {text}"
                )

            self.finish(self.phom_rankings())
            return

        self.turn = (seat + 1) % self.n
        self.phase = "draw"

        self.status = (
            f"{self.name(self.turn)}: ăn bài hoặc bốc bài."
        )

    def start_poker(self):
        self.deal(2)
        self.contrib = [0] * self.n
        self.street = 0
        self.limit = 20
        self.current = 20
        self.raise_count = 0
        self.pending = set(range(self.n))
        self.showdown = False

        if self.n == 2:
            small = self.button
            big = (self.button + 1) % self.n
        else:
            small = (self.button + 1) % self.n
            big = (self.button + 2) % self.n

        self.pay(small, 10)
        self.pay(big, 20)

        self.say(
            f"Button: {self.name(self.button)} | "
            f"SB {self.name(small)}: 10 | BB {self.name(big)}: 20"
        )
        self.poker_advance((big + 1) % self.n)

    def pay(self, seat, amount):
        # Mức cược cố định và giới hạn số lần tố đảm bảo đủ chip.
        if amount > self.chips[seat]:
            raise RuntimeError("Không đủ chip theo cấu hình fixed-limit.")

        self.chips[seat] -= amount
        self.bets[seat] += amount
        self.contrib[seat] += amount

    def action_poker(self, action):
        i = self.turn
        need = self.current - self.bets[i]

        if action == "fold":
            self.folded.add(i)
            self.say(f"{self.name(i)} bỏ bài.")

        elif action == "call":
            self.pay(i, need)
            self.say(
                f"{self.name(i)} theo {need} xu."
                if need else f"{self.name(i)} check."
            )

        elif action == "raise":
            if self.raise_count >= 3:
                raise ValueError("Vòng này đã đủ 3 lần tố.")

            self.current += self.limit
            self.raise_count += 1
            self.pay(i, self.current - self.bets[i])
            self.pending = set(range(self.n)) - self.folded - {i}

            self.say(f"{self.name(i)} tố tới {self.current} xu.")
        else:
            raise ValueError("Hành động Poker không hợp lệ.")

        self.pending.discard(i)
        self.poker_advance((i + 1) % self.n)

    def poker_advance(self, start):
        live = [i for i in range(self.n) if i not in self.folded]

        if len(live) == 1:
            winner = live[0]
            self.chips[winner] += sum(self.contrib)

            self.say(
                f"{self.name(winner)} nhận pot {sum(self.contrib)} xu "
                "không cần lật bài."
            )
            self.finish(self.groups_by(self.chips, reverse=True))
            return

        self.pending -= self.folded

        if not self.pending:
            if self.street == 3:
                self.poker_settle()
                return

            self.street += 1
            self.deck.pop()  # Burn card.

            count = 3 if self.street == 1 else 1
            self.board.extend(self.deck.pop() for _ in range(count))

            self.bets = [0] * self.n
            self.current = 0
            self.raise_count = 0
            self.limit = 20 if self.street == 1 else 40
            self.pending = set(live)
            start = (self.button + 1) % self.n

            label = ("", "FLOP", "TURN", "RIVER")[self.street]
            self.say(f"{label}: {cards_text(self.board)}")

        for step in range(self.n):
            i = (start + step) % self.n
            if i in self.pending:
                self.turn = i
                break

        label = ("Pre-flop", "Flop", "Turn", "River")[self.street]
        need = self.current - self.bets[self.turn]
        self.status = f"{label} • {self.name(self.turn)} • Cần theo {need} xu"

    def poker_settle(self):
        self.showdown = True

        scores = {
            i: poker_best(self.hands[i] + self.board)
            for i in range(self.n)
            if i not in self.folded
        }

        for i, score in scores.items():
            self.say(
                f"{self.name(i)}: {cards_text(self.hands[i])} "
                f"→ {POKER_NAMES[score[0]]}"
            )

        best = max(scores.values())
        winners = [i for i, score in scores.items() if score == best]

        # Chip lẻ ưu tiên ghế bên trái button.
        winners.sort(key=lambda i: (i - self.button - 1) % self.n)
        share, remainder = divmod(sum(self.contrib), len(winners))

        for position, i in enumerate(winners):
            self.chips[i] += share + (position < remainder)

        self.say(
            f"Pot {sum(self.contrib)} xu → "
            + ", ".join(self.name(i) for i in winners)
        )

        # Tournament Poker xếp hạng ván theo chip cuối ván.
        self.finish(self.groups_by(self.chips, reverse=True))

    def poker_strength(self, seat):
        """
        Heuristic rẻ để bot chạy nhanh trong giải 32 người.
        Chỉ đọc bài riêng, bài chung và số người chưa fold.
        """
        hole = self.hands[seat]
        opponents = self.n - len(self.folded) - 1

        if not self.board:
            a, b = sorted((c[0] for c in hole), reverse=True)
            score = 0.22 + (a + b - 4) / 60

            if a == b:
                score += 0.22
            if hole[0][1] == hole[1][1]:
                score += 0.06
            if abs(a - b) == 1:
                score += 0.035
        else:
            result = poker_best(hole + self.board)
            bases = (0.18, 0.40, 0.57, 0.67, 0.77, 0.83, 0.90, 0.96, 0.99)
            score = bases[result[0]] + result[1] / 250

            # Giá trị bộ nằm sẵn trên board giảm khi ai cũng dùng được.
            if len(self.board) == 5:
                board_result = poker_best(self.board)
                if result == board_result:
                    score *= 0.62

        score *= 1 / (1 + max(0, opponents - 1) * 0.13)
        return max(0.03, min(0.98, score))

    def start_cao(self):
        self.deal(3)
        self.revealed = set()
        self.pot = self.n * 25
        self.chips = [975] * self.n
        self.bets = [25] * self.n

        human = self.human_seat()
        self.turn = human if human is not None else 0
        self.status = "Mỗi ghế góp 25 xu. Bấm Lật bài để bắt đầu so bài."

    def action_cao(self, action):
        if action != "reveal":
            raise ValueError("Bấm Lật bài để tiếp tục.")

        i = self.turn
        self.revealed.add(i)
        self.say(f"{self.name(i)} lật bài: {cao_label(self.hands[i])}.")

        remaining = [j for j in range(self.n) if j not in self.revealed]

        if remaining:
            self.turn = remaining[0]
            self.status = f"Dealer chuẩn bị lật bài của {self.name(self.turn)}."
            return

        groups = self.groups_by(
            [cao_score(hand) for hand in self.hands],
            reverse=True,
        )

        winners = groups[0]
        share, remainder = divmod(self.pot, len(winners))

        for position, j in enumerate(winners):
            self.chips[j] += share + (position < remainder)

        self.finish(groups)

    def summary(self):
        if self.kind == "Poker":
            return (
                f"POT {sum(self.contrib):,} xu"
                f" • Fixed-limit {self.limit}\n"
                f"Tố {self.raise_count}/3 lần trong vòng"
            )

        if self.kind == "Ba cào":
            return (
                f"POT {self.pot} xu\n"
                "Ba Tây > 9 > 8 > ... > 0"
            )

        if self.kind == "Tá lả":
            return (
                f"Nọc: {len(self.deck)} lá"
                " • Mỗi người đánh 4 lượt\n"
                + " | ".join(
                    f"{self.name(seat)} {count}/4"
                    for seat, count in enumerate(
                        self.discard_count
                    )
                )
            )

        # Tiến lên.
        return (
            "3/4 đôi thông • Tứ quý • 4 đôi thông được rình\n"
            "Không về 2 • Cùng thối cùng hạng 4"
        )

    def analyze_base(self):
        seat = self.human_seat()

        if seat is None:
            return "Bạn đang quan sát bàn của các bot.", []

        if self.done:
            lines = []
            place = 1

            for group in self.ranking:
                label = ", ".join(self.name(i) for i in group)
                lines.append(f"Hạng {place}: {label}")
                place += len(group)

            return "\n".join(lines), []

        if self.kind == "Tiến lên":
            if seat in getattr(self, "thoi_two", []):
                return (
                    "Bạn đã bị tính THỐI 2 vì chỉ còn "
                    "các lá 2 trên tay.\n\n"
                    "Bạn không đánh tiếp trong ván này và "
                    "sẽ xếp sau những người về hợp lệ."
                ), []

            if not self.hands[seat]:
                return "Bạn đã hết bài. Đang chờ các ghế còn lại.", []

            if self.turn != seat:
                return (
                    "Chưa đến lượt bạn.\n\n"
                    "Ưu tiên giữ bộ/sảnh; tránh xé bộ chỉ để đè một lá nhỏ."
                ), []

            cards = self.tl_recommend(seat)

            if not cards:
                return (
                    "Gợi ý bỏ lượt.\n\n"
                    "Có thể do không có bộ đè hợp lệ, "
                    "cần tránh thối 2, hoặc đánh lúc này "
                    "sẽ phá cấu trúc bài quá nhiều."
                ), []

            remaining = self.tl_remaining_after(seat, cards)
            turns, singles = tl_plan(remaining)

            opponents = self.tl_dangerous_opponents(seat)

            one_card_threat = any(
                len(self.hands[other]) == 1
                and (
                    not self.table
                    or other not in self.passed
                )
                for other in opponents
            )

            reasons = []

            if not remaining:
                reasons.append("Đánh hết bài và về hợp lệ.")

            elif one_card_threat and len(cards) == 1:
                reasons.append(
                    "Ưu tiên lá lớn để chặn đối thủ còn một lá."
                )

            elif one_card_threat and len(cards) > 1:
                reasons.append(
                    "Dùng bộ nhiều lá để đối thủ một lá không thể đáp trả."
                )

            if (
                self.table
                and len(self.table) == 1
                and self.table[0][0] == 2
                and len(cards) == 4
            ):
                reasons.append(
                    "Dùng tứ quý chặt 2 để giành quyền mở dây."
                )

            if remaining:
                reasons.append(
                    f"Bài còn lại ước lượng chia được thành "
                    f"{turns} lượt, gồm {singles} lá đánh lẻ."
                )

            if any(card[0] == 2 for card in cards):
                reasons.append(
                    "Giải phóng 2 trước khi kết thúc, giảm nguy cơ thối."
                )

            explanation = "\n".join(
                "• " + reason for reason in reasons
            )

            return (
                f"Gợi ý: {cards_text(cards)}\n\n"
                f"{explanation}\n\n"
                "Đã chọn sẵn các lá gợi ý.\n\n"
                "AI xét cấu trúc bài, số lá đối thủ và bài đã công khai. "
                "Số lượt ước lượng không tính việc bị đối thủ đè. "
                "Không đọc bài úp và không bảo đảm nước đi tối ưu."
            ), cards

        if self.kind == "Tá lả":
            score, groups = phom_result(self.hands[seat], self.eaten[seat])
            text = "Phỏm hiện có:\n"
            text += "\n".join(cards_text(g) for g in groups) or "Chưa có phỏm"
            text += f"\n\nĐiểm bài rác: {score}"
            chosen = []

            if self.turn == seat:
                if self.phase == "draw":
                    text += (
                        "\n\nCó thể ăn lá vừa đánh."
                        if self.can_eat(seat)
                        else "\n\nKhông ăn được; nên bốc bài."
                    )
                else:
                    choices = self.phom_discards(seat)
                    if choices:
                        card = choices[0][2]
                        chosen = [card]
                        text += (
                            f"\n\nGợi ý đánh {card_text(card)}."
                            f"\nSau đánh: {choices[0][0]} điểm rác."
                        )

            text += "\n\nChưa ước lượng nguy cơ đối thủ ăn lá bạn đánh."
            return text, chosen

        if self.kind == "Ba cào":
            return (
                f"Bài: {cards_text(self.hands[seat])}\n"
                f"Kết quả: {cao_label(self.hands[seat])}\n\n"
                "A = 1; 2–9 theo số; 10/J/Q/K = 0.\n"
                "Lấy hàng đơn vị tổng điểm.\n"
                "Ba lá J/Q/K bất kỳ là Ba Tây.\n\n"
                "Không có quyết định chiến thuật sau khi chia."
            ), []

        if seat in self.folded:
            return "Bạn đã bỏ bài, đang quan sát ván.", []

        opponents = self.n - len(self.folded) - 1
        equity = equity_estimate(
            self.hands[seat],
            self.board,
            opponents,
            trials=70,
        )

        need = max(0, self.current - self.bets[seat])
        odds = need / (sum(self.contrib) + need) if need else 0

        if len(self.board) >= 3:
            category = POKER_NAMES[poker_best(self.hands[seat] + self.board)[0]]
        else:
            category = "Bài tẩy, chưa có bài chung"

        if self.turn != seat:
            advice = "Chưa đến lượt bạn."
        elif need and equity < odds:
            advice = "Cân nhắc bỏ bài: equity dưới pot odds."
        elif equity > 0.60 and self.raise_count < 3:
            advice = "Có thể cân nhắc tố."
        else:
            advice = "Ưu tiên check/theo nếu mức cược phù hợp."

        return (
            f"{category}\n"
            f"Đối thủ còn bài: {opponents}\n"
            f"Equity ước lượng: {equity:.1%}\n"
            f"Pot odds: {odds:.1%}\n"
            f"Cần theo: {need} xu\n\n"
            f"{advice}\n\n"
            "Monte Carlo 70 mẫu, sai số có thể lớn. "
            "Giả định bài đối thủ ngẫu nhiên, không suy luận range. "
            "Bot dùng heuristic nhẹ hơn để giải chạy nhanh."
        ), []

    def analyze(self):
        """
        Tiến lên dùng rank_map để hiển thị đúng hạng thực.
        Các trò khác tiếp tục dùng hàm phân tích cũ.
        """
        if self.kind == "Poker" and self.done:
            lines = [
                "KẾT QUẢ XU TRONG VÁN POKER",
                "",
            ]

            ordered = sorted(
                range(self.n),
                key=lambda seat: -self.chips[seat],
            )

            for seat in ordered:
                profit = self.chips[seat] - 1000

                lines.append(
                    f"{self.name(seat)}: "
                    f"{self.chips[seat]:,} xu "
                    f"({profit:+,})"
                )

            lines.append("")
            lines.append(
                "Đây là stack cuối ván. Trong Tournament, "
                "lãi/lỗ này được cộng vào số dư xu giải."
            )

            return "\n".join(lines), []

        if self.kind != "Tiến lên":
            return self.analyze_base()

        # ----------------------------------------------------
        # Ván đã kết thúc: hiển thị hạng thực.
        # ----------------------------------------------------
        if self.done:
            lines = [self.status, ""]

            ordered_seats = sorted(
                range(self.n),
                key=lambda seat: (
                    self.rank_map[seat],
                    seat,
                ),
            )

            for seat in ordered_seats:
                notes = []

                if seat in self.thoi_two:
                    notes.append("THỐI 2")

                if seat in self.white_labels:
                    notes.append(self.white_labels[seat])

                suffix = (
                    " — " + ", ".join(notes)
                    if notes
                    else ""
                )

                lines.append(
                    f"Hạng {self.rank_map[seat]}: "
                    f"{self.name(seat)}{suffix}"
                )

            return "\n".join(lines), []

        # ----------------------------------------------------
        # Ván đang diễn ra.
        # ----------------------------------------------------
        seat = self.human_seat()

        if seat is None:
            return "Bạn đang quan sát bàn bot.", []

        if seat in self.thoi_two:
            return (
                f"Bạn đã thối 2 và nhận hạng {self.n}.\n\n"
                "Bạn không đánh tiếp trong ván này. "
                "Những người khác vẫn tiếp tục nếu cần."
            ), []

        if seat in self.finished:
            rank = self.finished.index(seat) + 1

            return (
                f"Bạn đã về hợp lệ hạng {rank}.\n\n"
                "Đang chờ những người còn lại kết thúc."
            ), []

        if self.turn != seat:
            return (
                "Chưa đến lượt bạn.\n\n"
                "Bốn đôi thông có thể rình dù đã bỏ lượt. "
                "Game sẽ mở nút Chặt ngoài lượt khi bạn có "
                "cơ hội hợp lệ."
            ), []

        cards = self.tl_recommend(seat)

        if not cards:
            advice = (
                "Bỏ qua cơ hội rình."
                if self.tl_reacting
                else "Bỏ lượt."
            )

            return (
                f"Gợi ý: {advice}\n\n"
                "AI tránh thối 2 và hạn chế phá cấu trúc bài."
            ), []

        remaining = self.tl_remaining_after(seat, cards)
        turns, singles = tl_plan(remaining)

        header = (
            "⚡ Gợi ý RÌNH"
            if self.tl_reacting
            else "Gợi ý đánh"
        )

        return (
            f"{header}: {cards_text(cards)}\n\n"
            f"Bài còn lại: {len(remaining)} lá.\n"
            f"Ước lượng: {turns} lượt đánh, "
            f"{singles} lá đánh lẻ.\n\n"
            "Đã chọn sẵn bài gợi ý.\n\n"
            "AI xét bài công khai, cấu trúc bài và số lá đối thủ; "
            "không nhìn bài úp. Ước lượng chưa tính mọi khả năng bị đè."
        ), cards

class Tournament:
    """
    32 -> 16 -> 8 -> Final Table 4.
    Mỗi bàn 4 người, 3 ván.
    Điểm mỗi ván: 4 / 3 / 2 / 1.
    Đồng hạng chia trung bình số điểm các vị trí tương ứng.
    Tie-break bàn: điểm -> số lần nhất -> hạt giống bốc thăm.
    """

    def __init__(self, kind):
        self.kind = kind
        self.entrants = list(range(32))

        draw = list(range(32))
        random.shuffle(draw)
        self.seed = {pid: index for index, pid in enumerate(draw)}

        self.total_points = {pid: 0.0 for pid in range(32)}
        self.eliminated = []
        self.history = []
        # Poker dùng số dư xu, không dùng điểm thứ hạng.
        self.poker_starting_bank = 1000

        self.poker_bank = {
            pid: self.poker_starting_bank
            for pid in range(32)
        }

        self.poker_hands_played = {
            pid: 0
            for pid in range(32)
        }
        self.complete = False
        self.final_order = []
        self.start_stage()

    def start_stage(self):
        self.stage_size = len(self.entrants)

        ids = list(self.entrants)
        random.shuffle(ids)
        self.groups = [ids[i:i + 4] for i in range(0, len(ids), 4)]

        # Đưa bàn có người chơi thật lên trước.
        self.groups.sort(key=lambda group: 0 not in group)

        self.group_index = 0
        self.qualifiers = []
        self.stage_losers = []
        self.reset_match()

    def reset_match(self):
        self.hand_index = 0

        if self.kind == "Poker":
            # Giữ số dư xu đã tích lũy từ các vòng trước.
            #
            # self.points được giữ làm trường nội bộ chung,
            # nhưng với Poker, giá trị bên trong là XU,
            # không phải điểm hạng.
            self.points = {
                pid: self.poker_bank[pid]
                for pid in self.current_group
            }
        else:
            self.points = {
                pid: 0.0
                for pid in self.current_group
            }

        self.wins = {
            pid: 0
            for pid in self.current_group
        }

    @property
    def current_group(self):
        return self.groups[self.group_index]

    @property
    def label(self):
        return "FINAL TABLE" if self.stage_size == 4 else f"VÒNG {self.stage_size}"

    def record_hand(self, engine):
        # ====================================================
        # POKER: GHI LÃI/LỖ XU, KHÔNG TÍNH ĐIỂM THỨ HẠNG
        # ====================================================
        if engine.kind == "Poker":
            if not engine.done:
                raise ValueError(
                    "Chỉ được ghi kết quả Poker khi ván đã kết thúc."
                )

            # Engine Poker hiện tại luôn bắt đầu mỗi ghế 1.000 xu.
            hand_starting_stack = 1000

            for seat, pid in enumerate(engine.ids):
                ending_stack = engine.chips[seat]
                profit = ending_stack - hand_starting_stack

                self.poker_bank[pid] += profit
                self.poker_hands_played[pid] += 1

                # Dùng xu làm tiêu chí xét đi tiếp.
                self.points[pid] = self.poker_bank[pid]

            return

        # ====================================================
        # TIẾN LÊN: GIỮ HẠNG THỰC, CÙNG THỐI CÙNG HẠNG 4
        # ====================================================
        explicit_ranks = getattr(engine, "rank_map", {})

        if engine.kind == "Tiến lên" and explicit_ranks:
            for seat, rank in explicit_ranks.items():
                pid = engine.ids[seat]
                points = 5 - rank

                self.points[pid] += points
                self.total_points[pid] += points

                if rank == 1:
                    self.wins[pid] += 1

            return

        # ====================================================
        # CÁC GAME KHÁC: GIỮ CÁCH TÍNH ĐIỂM CŨ
        # ====================================================
        position = 0

        for group in engine.ranking:
            values = [
                4 - rank
                for rank in range(
                    position,
                    position + len(group),
                )
            ]

            average = sum(values) / len(values)

            for seat in group:
                pid = engine.ids[seat]

                self.points[pid] += average
                self.total_points[pid] += average

                if position == 0:
                    self.wins[pid] += 1

            position += len(group)

    def match_order(self):
        if self.kind == "Poker":
            return sorted(
                self.current_group,
                key=lambda pid: (
                    -self.poker_bank[pid],
                    self.seed[pid],
                ),
            )

        return sorted(
            self.current_group,
            key=lambda pid: (
                -self.points[pid],
                -self.wins[pid],
                self.seed[pid],
            ),
        )

    def advance(self):
        """Gọi sau khi ván hiện tại đã được ghi điểm."""
        self.hand_index += 1

        if self.hand_index < 3:
            return

        order = self.match_order()

        self.history.append({
            "stage": self.stage_size,
            "order": list(order),
            "points": dict(self.points),
            "wins": dict(self.wins),
        })

        if self.stage_size == 4:
            self.final_order = order + self.eliminated
            self.complete = True
            return

        self.qualifiers.extend(order[:2])

        for pid in order[2:]:
            self.stage_losers.append(
                (pid, self.points[pid], self.wins[pid])
            )

        self.group_index += 1

        if self.group_index < len(self.groups):
            self.reset_match()
            return

        # Người bị loại ở vòng sâu hơn được xếp cao hơn vòng trước.
        losers = sorted(
            self.stage_losers,
            key=lambda entry: (-entry[1], -entry[2], self.seed[entry[0]]),
        )
        self.eliminated = [entry[0] for entry in losers] + self.eliminated

        self.entrants = list(self.qualifiers)
        self.start_stage()

    def report_base(self):
        lines = [
            f"TOURNAMENT — {self.kind}",
            "32 → 16 → 8 → Final Table 4",
            "Mỗi bàn 3 ván; 2 người đầu đi tiếp.",
            "Điểm ván: 4 / 3 / 2 / 1.",
            "Tiến lên: tính theo hạng thực; cùng thối cùng hạng 4.",
            "Các trò khác: đồng hạng chia trung bình điểm.",
            "Phá hòa: điểm → số lần nhất → hạt giống bốc thăm.",
            "",
        ]

        if self.complete:
            lines.append("BẢNG XẾP HẠNG CUỐI CÙNG")
            lines.append(
                "Ưu tiên vòng tiến sâu nhất, sau đó kết quả bàn."
            )
            lines.append(
                "Hạng 1–4 được quyết định bằng điểm Final."
            )
            lines.append("")

            # Lấy kết quả bàn cuối cùng mà mỗi người tham gia.
            last_result = {}

            for match in self.history:
                for pid in match["order"]:
                    last_result[pid] = {
                        "stage": match["stage"],
                        "points": match["points"][pid],
                        "wins": match["wins"][pid],
                    }

            for rank, pid in enumerate(self.final_order, 1):
                result = last_result.get(pid)

                if result is None:
                    stage_label = "Chưa rõ"
                    deciding_points = 0.0
                    deciding_wins = 0
                else:
                    stage_label = (
                        "Final"
                        if result["stage"] == 4
                        else f"Vòng {result['stage']}"
                    )

                    deciding_points = result["points"]
                    deciding_wins = result["wins"]

                marker = "  ← BẠN" if pid == 0 else ""

                lines.append(
                    f"{rank:>2}. {player_name(pid):<13}"
                    f" | {stage_label:<7}"
                    f" | Điểm bàn {deciding_points:4.1f}"
                    f" | Nhất {deciding_wins}"
                    f" | Tổng {self.total_points[pid]:5.1f}"
                    f"{marker}"
                )

            lines.append("")
            lines.append("CÁCH XẾP HẠNG")
            lines.append(
                "• Vào vòng sâu hơn luôn được xếp cao hơn."
            )
            lines.append(
                "• Hạng 1–4: điểm Final → số lần nhất Final "
                "→ hạt giống."
            )
            lines.append(
                "• Cùng vòng bị loại: điểm bàn → số lần nhất "
                "→ hạt giống."
            )
            lines.append(
                "• Tổng điểm toàn giải không quyết định thứ hạng."
            )
        else:
            lines.append(
                f"{self.label} • Bàn {self.group_index + 1}/{len(self.groups)}"
            )
            lines.append(f"Ván {self.hand_index + 1}/3")
            lines.append("")

            for pid in self.match_order():
                lines.append(
                    f"{player_name(pid)}: {self.points[pid]:.1f} điểm, "
                    f"{self.wins[pid]} lần nhất"
                )

            lines.append(f"\nĐã loại: {len(self.eliminated)} người.")

        lines.append("\nLỊCH SỬ CÁC BÀN")
        for match in self.history:
            stage = "FINAL" if match["stage"] == 4 else f"Vòng {match['stage']}"
            description = " > ".join(
                f"{player_name(pid)} ({match['points'][pid]:.1f})"
                for pid in match["order"]
            )
            lines.append(f"{stage}: {description}")

        return "\n".join(lines)
    def report(self):
        if self.kind != "Poker":
            return self.report_base()

        lines = [
            "POKER — BẢNG XU GIẢI",
            "",
            "Không sử dụng điểm hạng 4 / 3 / 2 / 1.",
            "Số dư ban đầu: 1.000 xu/người.",
            "Xu giải được cộng/trừ theo lãi/lỗ mỗi ván.",
            "Mỗi bàn 3 ván; hai người nhiều xu nhất đi tiếp.",
            "Bằng xu khi xét đi tiếp: dùng hạt giống bốc thăm.",
            "",
        ]

        # Ghi nhận vòng gần nhất mỗi người đã hoàn tất.
        last_stage = {}

        for match in self.history:
            for pid in match["order"]:
                last_stage[pid] = match["stage"]

        if self.complete:
            lines.append("GIẢI ĐÃ HOÀN TẤT")
        else:
            lines.append(
                f"{self.label}"
                f" • Bàn {self.group_index + 1}/{len(self.groups)}"
                f" • Ván {self.hand_index + 1}/3"
            )

            lines.append("")
            lines.append("XU TẠI BÀN HIỆN TẠI")

            for pid in self.match_order():
                marker = " ← BẠN" if pid == 0 else ""

                lines.append(
                    f"{player_name(pid):<14}"
                    f" | {self.poker_bank[pid]:>7,} xu"
                    f"{marker}"
                )

        lines.append("")
        lines.append("SỐ DƯ TOÀN GIẢI — SẮP THEO XU")
        lines.append(
            "Tên            | Xu hiện có |  Lãi/lỗ | Số ván | Trạng thái"
        )
        lines.append("-" * 79)

        ordered = sorted(
            range(32),
            key=lambda pid: (
                -self.poker_bank[pid],
                self.seed[pid],
            ),
        )

        active_ids = set(self.entrants)

        if not self.complete:
            active_ids.update(self.qualifiers)

        for pid in ordered:
            bank = self.poker_bank[pid]
            profit = bank - self.poker_starting_bank
            played = self.poker_hands_played[pid]

            if self.complete:
                stage = last_stage.get(pid)

                if stage == 4:
                    status = "Final Table"
                elif stage is not None:
                    status = f"Dừng ở vòng {stage}"
                else:
                    status = "Chưa có kết quả"

            elif pid in active_ids:
                status = "Đang tham gia"
            else:
                status = "Đã bị loại"

            marker = " ← BẠN" if pid == 0 else ""

            lines.append(
                f"{player_name(pid):<14}"
                f" | {bank:>10,}"
                f" | {profit:>+7,}"
                f" | {played:>6}"
                f" | {status}"
                f"{marker}"
            )

        lines.append("")
        lines.append(
            "Bảng này sắp theo số dư xu, không đánh số thứ hạng."
        )
        lines.append(
            "Trạng thái cho biết người chơi tiến tới vòng nào."
        )
        lines.append(
            "Người bị loại vẫn có thể có số dư cao hơn người "
            "chơi tiếp nhưng thua xu ở các vòng sau."
        )

        lines.append("")
        lines.append("LỊCH SỬ SỐ DƯ SAU MỖI BÀN")

        for match in self.history:
            stage_label = (
                "FINAL"
                if match["stage"] == 4
                else f"Vòng {match['stage']}"
            )

            entries = [
                f"{player_name(pid)}: "
                f"{match['points'][pid]:,.0f} xu"
                for pid in match["order"]
            ]

            lines.append(
                stage_label + " | " + " • ".join(entries)
            )

        return "\n".join(lines)

    def cancel_timer(self):
        """Hủy lịch hiện tại và xóa thông tin callback."""
        if self.timer is not None:
            try:
                self.after_cancel(self.timer)
            except tk.TclError:
                pass

        self.timer = None
        self.timer_callback = None
        self.timer_due = None
        self.timer_remaining = None

    def schedule(self, function, delay):
        """
        Lên lịch một bước của game.

        Nếu đang tạm dừng, chỉ lưu callback và thời gian chờ,
        chưa thực thi.
        """
        self.cancel_timer()

        self.timer_callback = function
        self.timer_remaining = max(0, int(delay))

        if not self.paused:
            self.arm_timer()

    def arm_timer(self):
        """Khởi động callback đang chờ bằng thời gian còn lại."""
        if self.paused or self.timer_callback is None:
            return

        delay = max(0, int(self.timer_remaining or 0))
        self.timer_due = time.monotonic() + delay / 1000.0

        def callback():
            self.timer = None

            if self.paused:
                # Nhánh bảo vệ: không chạy bước game khi đang dừng.
                self.timer_remaining = max(
                    0,
                    int(
                        ((self.timer_due or time.monotonic())
                         - time.monotonic()) * 1000
                    ),
                )
                self.timer_due = None
                return

            function = self.timer_callback

            self.timer_callback = None
            self.timer_due = None
            self.timer_remaining = None

            if function is not None:
                function()

        self.timer = self.after(delay, callback)

    def toggle_pause(self, event=None):
        """F8 hoặc nút trên thanh tiêu đề."""
        if self.game is None:
            return "break"

        if self.tournament and self.tournament.complete:
            return "break"

        if not self.paused:
            self.paused = True

            # Giữ lại callback, chỉ hủy lịch thực thi của Tkinter.
            if self.timer is not None:
                if self.timer_due is not None:
                    self.timer_remaining = max(
                        0,
                        int(
                            (self.timer_due - time.monotonic()) * 1000
                        ),
                    )

                try:
                    self.after_cancel(self.timer)
                except tk.TclError:
                    pass

                self.timer = None
                self.timer_due = None

            self.write("⏸ Đã tạm dừng. Bấm F8 hoặc Tiếp tục để chơi tiếp.")
            self.render()

        else:
            self.paused = False
            self.write("▶ Tiếp tục chơi.")

            if self.timer_callback is not None:
                # Tiếp tục đúng callback và thời gian chờ trước khi dừng.
                self.arm_timer()
                self.render()

            elif self.silent and not self.game.done:
                # Bàn bot của tournament đang mô phỏng.
                self.schedule(self.simulate_table_hand, 10)
                self.render()

            else:
                # Có thể đang ở lượt người chơi hoặc đã hết ván.
                self.pump()

        return "break"

    def new_session(self, confirm=True):

        self.paused = False

        self.bubbles.clear()

        self.log_box.delete("1.0", "end")
        self.log_box.configure(state="disabled")

        self.set_analysis("Bấm Phân tích AI để phân tích tay bài hiện tại.")

        if self.play_mode.get() == "Tournament 32":
            self.tournament = Tournament(self.kind.get())
            self.write("🏆 Bắt đầu giải 32 người. Mỗi bàn 3 ván, lấy top 2.")
        else:
            self.tournament = None

        self.start_hand()

    def start_hand(self):
        self.selected.clear()
        self.bubbles.clear()
        self.accounted = False

        if self.tournament:
            t = self.tournament

            if t.complete:
                self.render()
                self.show_standings()
                return

            # Luân phiên ghế mở bài giữa 3 ván của cùng bàn.
            ids = list(t.current_group)
            shift = t.hand_index % 4
            ids = ids[shift:] + ids[:shift]
            kind = t.kind
            hand_number = 0
            label = (
                f"{t.label} • Bàn {t.group_index + 1}/{len(t.groups)} "
                f"• Ván {t.hand_index + 1}/3"
            )
        else:
            kind = self.kind.get()
            n = int(self.seats.get()) if kind in ("Poker", "Ba cào") else 4
            ids = list(range(n))
            hand_number = self.quick_hand
            label = f"Ván nhanh #{self.quick_hand + 1}"

        self.silent = 0 not in ids
        self.write("\n" + "━" * 25)
        self.write(label)

        self.game = Engine(
            kind, ids, hand_number,
            logger=self.engine_log,
        )

        # Liên kết điểm giải cho Tiến lên và Tá lả.
        # Ván nhanh không có đối tượng Tournament.
        self.game.score_tournament = (
            self.tournament
            if (
                self.tournament is not None
                and kind in ("Tiến lên", "Tá lả")
            )
            else None
        )

        self.set_analysis(
            "Bàn tự động đang được mô phỏng..."
            if self.silent else
            "Bấm Phân tích AI để xem bài và gợi ý."
        )

        self.render()

        if self.silent:
            self.schedule(self.simulate_table_hand, 100)
        else:
            self.pump()

    def engine_log(self, text):
        if not self.silent:
            self.write(text)

    def simulate_table_hand(self):
        """
        Mỗi callback chỉ chạy một hành động bot.
        Sau đó trả quyền xử lý sự kiện cho giao diện.
        """
        if self.paused:
            return

        game = self.game

        if game is None:
            return

        # Hàm này chỉ chạy bàn không có người thật.
        if not self.silent:
            self.pump()
            return

        if game.done:
            self.pump()
            return

        try:
            steps = getattr(game, "_simulation_steps", 0)

            if steps >= 2500:
                raise RuntimeError(
                    "Bàn mô phỏng vượt 2500 hành động. "
                    "Có thể engine đang lặp trạng thái."
                )

            if not 0 <= game.turn < game.n:
                raise RuntimeError(
                    f"Lượt bot không hợp lệ: turn={game.turn}, "
                    f"done={game.done}, game={game.kind}"
                )

            seat = game.turn
            name = game.name(seat)
            started = time.monotonic()

            # Thực hiện đúng một hành động.
            game.bot()

            elapsed = time.monotonic() - started
            game._simulation_steps = steps + 1

            # In tiến trình để biết bàn có đang chạy hay không.
            if steps == 0:
                self.write(
                    f"🤖 Bàn mô phỏng bắt đầu: {name} đã hành động."
                )

            if (steps + 1) % 20 == 0:
                counts = " | ".join(
                    f"{game.name(i)}: {len(game.hands[i])} lá"
                    for i in range(game.n)
                )

                self.write(
                    f"Mô phỏng bước {steps + 1}: {counts}"
                )

            if elapsed > 0.5:
                self.write(
                    f"⏱ {name} tính một nước mất "
                    f"{elapsed:.2f} giây."
                )

        except Exception:
            error = traceback.format_exc()

            self.cancel_timer()
            self.paused = True

            print(error)

            self.write("❌ Bàn mô phỏng gặp lỗi và đã tạm dừng.")
            self.set_analysis(
                "LỖI BÀN MÔ PHỎNG\n\n"
                + error
                + "\nHãy sao chép lỗi này để kiểm tra. "
                  "Không nên tiếp tục ván lỗi."
            )

            try:
                with open(
                    "card_lounge_error.log",
                    "a",
                    encoding="utf-8",
                ) as file:
                    file.write(
                        "\n"
                        + "=" * 60
                        + "\n"
                        + time.strftime("%Y-%m-%d %H:%M:%S")
                        + "\n"
                        + error
                    )
            except OSError:
                pass

            self.render()
            return

        self.render()

        if game.done:
            self.pump()
        else:
            # Tiếp tục bàn hiện tại, không tạo ván mới.
            self.schedule(self.simulate_table_hand, 30)

    def pump(self):
        if self.paused:
            self.render()
            return

        game = self.game

        if game is None:
            return

        if game.done:
            if not self.accounted:
                self.accounted = True

                if self.tournament:
                    self.tournament.record_hand(game)

                if self.silent:
                    self.write(game.status)

                if self.sound.get() and not self.silent:
                    self.bell()

                if not self.silent:
                    text, _ = game.analyze()
                    self.set_analysis(text)

            self.render()

            if self.silent:
                self.schedule(self.next_hand, 120)

            return

        # Bàn không có người thật vẫn chạy theo từng đợt nhỏ.
        if self.silent:
            self.render()
            self.schedule(self.simulate_table_hand, 30)
            return

        human_turn = game.ids[game.turn] == 0

        human_reaction = (
            human_turn
            and game.kind == "Tiến lên"
            and game.tl_reacting
        )

        if human_reaction:
            suggested = game.tl_recommend(game.turn)
            self.selected = set(suggested)

            self.set_analysis(
                "⚡ CƠ HỘI RÌNH — 4 GIÂY\n\n"
                "Bạn được dùng 4 đôi thông để chặt, "
                "kể cả đã bỏ lượt.\n\n"
                "Đã chọn sẵn bộ AI gợi ý nếu có.\n"
                "Bấm Chặt ngoài lượt hoặc Bỏ qua rình.\n\n"
                "F8 tạm dừng cả đồng hồ này."
            )

            self.render()

            expected_game = game
            expected_seat = game.turn

            def timeout():
                if (
                    self.game is expected_game
                    and not expected_game.done
                    and expected_game.tl_reacting
                    and expected_game.turn == expected_seat
                    and expected_game.ids[expected_seat] == 0
                ):
                    self.human_action("skip_cut")

            self.schedule(timeout, 4000)
            return

        self.render()

        if not human_turn:
            base = {
                "Chậm": 1450,
                "Vừa": 850,
                "Nhanh": 250,
            }[self.speed.get()]

            self.schedule(
                self.bot_step,
                base + random.randint(0, 200),
            )

    def bot_step(self):
        if self.paused:
            return

        game = self.game

        if not game or game.done:
            return

        pid = game.ids[game.turn]

        try:
            game.bot()
        except Exception as exc:
            messagebox.showerror("Lỗi bot", str(exc))
            return

        if random.random() < 0.20:
            self.chat(pid, random.choice(PHRASES + EMOJIS))

        self.pump()

    def human_action(self, action):
        if self.paused:
            return

        game = self.game

        if (
            game is None
            or game.done
            or game.turn < 0
            or game.ids[game.turn] != 0
        ):
            return

        try:
            game.action(action, tuple(self.selected))

        except ValueError as exc:
            # Không dùng hộp thoại modal trong cửa sổ rình:
            # tránh gây nhập nhằng với đồng hồ đang chạy.
            if (
                game.kind == "Tiến lên"
                and game.tl_reacting
            ):
                self.set_analysis(
                    "Chưa chặt được:\n\n"
                    + str(exc)
                    + "\n\nĐồng hồ rình vẫn đang chạy. "
                    "Bạn có thể bấm F8 để tạm dừng."
                )
            else:
                messagebox.showinfo(
                    "Nước đi chưa hợp lệ",
                    str(exc),
                )
            return

        self.cancel_timer()
        self.selected.clear()
        self.pump()

    def next_hand(self):
        if self.paused:
            return

        if not self.game or not self.game.done:
            return

        self.cancel_timer()

        if self.tournament:
            t = self.tournament
            previous_size = t.stage_size
            previous_group = list(t.current_group)
            finishing_match = t.hand_index == 2

            if finishing_match:
                order = t.match_order()

                if t.kind == "Poker":
                    self.write(
                        "Số dư Poker sau bàn: "
                        + " | ".join(
                            f"{player_name(pid)}: "
                            f"{t.poker_bank[pid]:,} xu"
                            for pid in order
                        )
                    )
                else:
                    self.write(
                        "Xếp hạng bàn: "
                        + " > ".join(
                            f"{player_name(pid)} "
                            f"({t.points[pid]:.1f} điểm)"
                            for pid in order
                        )
                    )

                if 0 in previous_group and previous_size > 4:
                    if 0 in order[:2]:
                        self.write("🎉 Bạn đi tiếp!")
                    else:
                        self.write("Bạn đã bị loại. Giải tiếp tục mô phỏng các bàn còn lại.")

            t.advance()

            if t.complete:
                self.write("🏆 TOURNAMENT HOÀN TẤT!")
                self.render()
                self.show_standings()
                return

            if t.stage_size != previous_size:
                self.write(f"\n🔥 Bắt đầu {t.label}!")

        else:
            self.quick_hand += 1

        self.start_hand()

    def write(self, text, tag=None):
        self.log_box.configure(state="normal")
        if tag:
            self.log_box.insert("end", text + "\n", tag)
        else:

    def chat(self, pid, text):
        self.write(f"{player_name(pid)}: {text}", "chat")

        token = object()
        self.bubbles[pid] = (text, token)
        self.render()

        def remove():
            current = self.bubbles.get(pid)
            if current and current[1] is token:
                self.bubbles.pop(pid, None)
                self.render()

        self.after(3600, remove)

    def set_analysis(self, text):
        self.analysis_box.configure(state="normal")
        self.analysis_box.delete("1.0", "end")
        self.analysis_box.insert("1.0", text)
        self.analysis_box.configure(state="disabled")

    def analyze(self):
        if not self.game:
            return

        self.configure(cursor="watch")
        self.update_idletasks()

        try:
            text, cards = self.game.analyze()
            self.set_analysis(text)
            if cards:
                self.selected = set(cards)
        finally:
            self.configure(cursor="")

        self.render()

    def clear_selection(self):
        self.selected.clear()
        self.render()

    def toggle_card(self, card):
        if getattr(self, "paused", False):
            return

        game = self.game

        if game is None or game.done:
            return

        if game.kind == "Tá lả":
            seat = game.human_seat()

            if seat is None:
                return

            if (
                game.turn != seat
                or game.phase != "discard"
                or card not in game.phom_loose[seat]
            ):
                return

            # Tá lả mỗi lần chỉ đánh một lá.
            if card in self.selected:
                self.selected.clear()
            else:
                self.selected.clear()
                self.selected.add(card)

        else:
            # Giữ cách chọn nhiều lá của Tiến lên.
            if card in self.selected:
                self.selected.remove(card)
            else:
                self.selected.add(card)

        self.render()

    def seat_details(self, seat):
        game = self.game

        # ----------------------------------------------------
        # TIẾN LÊN
        # ----------------------------------------------------
        if game.kind == "Tiến lên":
            # Đã kết thúc: dùng hạng thực từ engine.
            if game.done and getattr(game, "rank_map", {}):
                suffix = (
                    " • THỐI 2"
                    if seat in game.thoi_two
                    else ""
                )

                return (
                    f"Hạng {game.rank_map[seat]}"
                    f"{suffix}"
                )

            # Đang chơi nhưng ghế này đã thối.
            if seat in game.thoi_two:
                return f"THỐI 2 • Hạng {game.n}"

            # Đã về hợp lệ, đang chờ các ghế khác.
            if seat in game.finished:
                rank = game.finished.index(seat) + 1
                return f"Đã về hạng {rank}"

            return f"Còn {len(game.hands[seat])} lá"

        # ----------------------------------------------------
        # POKER
        # ----------------------------------------------------
        if game.kind == "Poker":
            if seat in game.folded:
                return (
                    f"FOLD • {game.chips[seat]:,} xu"
                )

            return (
                f"{game.chips[seat]:,} xu"
                f" • Cược {game.bets[seat]}"
            )

        # ----------------------------------------------------
        # BA CÀO
        # ----------------------------------------------------
        if game.kind == "Ba cào":
            if game.visible(seat):
                return (
                    f"{cao_label(game.hands[seat])}"
                    f" • {game.chips[seat]:,} xu"
                )

            return (
                f"{game.chips[seat]:,} xu"
                " • Góp 25"
            )

        # ----------------------------------------------------
        # TÁ LẢ
        # ----------------------------------------------------
        if game.kind == "Tá lả":
            return (
                f"{len(game.hands[seat])} lá"
                f" • Đánh {game.discard_count[seat]}/4"
            )

        return ""

    def draw_phom_human(self, seat, width, height):
        """Vẽ bài rác/cạ chờ và phỏm thành hai vùng riêng."""
        canvas = self.canvas
        game = self.game

        active = game.turn == seat and not game.done

        selectable = (
            active
            and game.phase == "discard"
            and not getattr(self, "paused", False)
        )

        loose = game.phom_loose[seat]
        melds = game.phom_melds[seat]
        eaten = set(game.eaten[seat])

        # Chỉ bài ngoài phỏm mới được chọn đánh.
        self.selected.intersection_update(loose)

        if not selectable:
            self.selected.clear()

        top = height - 139
        bottom = height - 5

        canvas.create_rectangle(
            12, top,
            width - 12, bottom,
            fill="#101b2a",

        )

        canvas.create_text(
            25, top + 13,
            text=label,
            anchor="w",
            fill="#ffe19b",

            text=(

            anchor="e",
            fill="#cbd9eb",

        if melds:
            split = width * 0.53

            canvas.create_line(
                split, top + 28,

                fill="#40516a",
            )

            loose_left = 25
            loose_right = split - 12
            meld_left = split + 12
            meld_right = width - 25
        else:
            loose_left = 25
            loose_right = width - 25

                list(pair)

                    card_height,

                )

            canvas.create_text(
                (loose_left + loose_right) / 2,
                card_y + 35,

                font=("Segoe UI", 11, "bold"),

                anchor="w",

                font=("Segoe UI", 9, "bold"),

            cursor = 0.0

            for group_index, group in enumerate(melds):

                    cursor += 0.5

                for card in group:
                    slots.append((card, cursor))
                    cursor += 1.0

            available = meld_right - meld_left
            last_slot = slots[-1][1]

            step = min(
                47,

                    1,
                    (available - card_width)
                    / max(1, last_slot),
                ),

            for card, position in slots:
                x = meld_left + position * step

                # Chủ bài nhìn thấy toàn bộ phỏm của mình.
                # Không gắn tag nên không chọn đánh được.

    def render_controls(self):
        for widget in self.actions.winfo_children():
            widget.destroy()

        game = self.game

        if game is None:
            self.next_button.configure(state="disabled")
            return

        is_human_turn = (
            not game.done
            and game.turn >= 0
            and game.ids[game.turn] == 0
        )

        for label, action in game.options():
            button = ttk.Button(
                self.actions,
                text=label,
                command=lambda value=action: self.human_action(value),
            )
            button.pack(side="left", padx=3)

            enabled = is_human_turn

            if (
                enabled
                and game.kind == "Poker"
                and action == "raise"
                and game.raise_count >= 3
            ):
                enabled = False

            if (
                enabled
                and game.kind == "Tá lả"
                and action == "eat"
                and not game.can_eat(game.turn)
            ):
                enabled = False

            if (
                enabled
                and game.kind == "Tiến lên"
                and action == "pass"
                and not game.table
            ):
                enabled = False

            button.configure(state="normal" if enabled else "disabled")

        if game.done:
            ttk.Label(
                self.actions,
                text="✓ Ván đã kết thúc",
            ).pack(side="left", padx=7)

        elif not is_human_turn:
            ttk.Label(
                self.actions,
                text="Bot đang suy nghĩ...",
            ).pack(side="left", padx=7)

        tournament = self.tournament

        if tournament and tournament.complete:
            self.next_button.configure(
                text="🏆 Xem bảng xếp hạng cuối cùng",
                command=self.show_standings,
                state="normal",
            )
            return

        self.next_button.configure(command=self.next_hand)

        if not game.done:
            self.next_button.configure(
                text="Chờ ván hiện tại kết thúc",
                state="disabled",
            )
            return

        if self.silent:
            self.next_button.configure(
                text="Đang tự động chuyển ván...",
                state="disabled",
            )
            return

        if tournament:
            if tournament.hand_index == 2:
                label = (
                    "Hoàn tất Final Table → Xem xếp hạng"
                    if tournament.stage_size == 4
                    else "Chốt kết quả bàn → Tiếp tục giải"
                )
            else:
                label = (
                    f"Tiếp tục → Ván {tournament.hand_index + 2}/3"
                )
        else:
            label = "Chơi ván nhanh tiếp theo"

        self.next_button.configure(
            text=label,
            state="normal",
        )

    def engine_init(
        self,
        kind,
        ids,
        hand_number=0,
        logger=None,
        starting_stacks=None,
    ):
        self._poker_initial_stacks = (
            list(starting_stacks)
            if starting_stacks is not None
            else None
        )

        old_engine_init(
            self,
            kind,
            ids,
            hand_number,
            logger,
        )

    def start_poker(self):
        self.deal(2)

        stacks = self._poker_initial_stacks

        if stacks is None:
            stacks = [1000] * self.n

        if len(stacks) != self.n:
            raise ValueError("Số stack không khớp số ghế.")

        if self.n < 2:
            raise ValueError("Cần ít nhất hai người để mở ván Poker.")

        if any(
            not isinstance(value, int) or value <= 0
            for value in stacks
        ):
            raise ValueError("Stack đầu ván phải là số nguyên dương.")

        self.chips = list(stacks)
        self.starting_chips = list(stacks)

        self.contrib = [0] * self.n
        self.bets = [0] * self.n
        self.folded = set()
        self.all_in = set()

        self.street = 0
        self.current = 20
        self.min_raise = 20

        # Giữ trường này để tương thích một số code giao diện cũ.
        self.limit = 20
        self.raise_count = 0

        self.last_action_level = [None] * self.n
        self.last_action_kind = [None] * self.n

        self.showdown = False
        self.side_pots = []
        self.hand_values = {}
        self.hand_winners = []

        if self.n == 2:
            self.small_blind = self.button
            self.big_blind = (self.button + 1) % self.n
        else:
            self.small_blind = (self.button + 1) % self.n
            self.big_blind = (self.button + 2) % self.n

        poker_pay(self, self.small_blind, 10)
        poker_pay(self, self.big_blind, 20)

        self.pending = {
            seat
            for seat in range(self.n)
            if self.chips[seat] > 0
        }

        self.say(
            f"Button: {self.name(self.button)} | "
            f"SB: {self.name(self.small_blind)} "
            f"({self.bets[self.small_blind]} xu) | "
            f"BB: {self.name(self.big_blind)} "
            f"({self.bets[self.big_blind]} xu)"
        )

        poker_progress(
            self,
            (self.big_blind + 1) % self.n,
        )

    def poker_pay(self, seat, amount):
        amount = max(0, min(int(amount), self.chips[seat]))

        self.chips[seat] -= amount
        self.bets[seat] += amount
        self.contrib[seat] += amount

        if self.chips[seat] == 0:
            self.all_in.add(seat)

        return amount

    def live_seats(self):
        return [
            seat
            for seat in range(self.n)
            if seat not in self.folded
        ]

    def can_raise(self, seat):
        need = max(0, self.current - self.bets[seat])

        if self.chips[seat] <= need:
            return False

        # Không tố vào một bàn mà mọi đối thủ đều đã all-in.
        if not any(
            other != seat
            and other not in self.folded
            and self.chips[other] > 0
            for other in range(self.n)
        ):
            return False

        previous = self.last_action_level[seat]

        if previous is None:
            return True

        # Người đã check vẫn được đáp lại một cược mở mới.
        if (
            self.last_action_kind[seat] == "check"
            and self.current > previous
        ):
            return True

        # Một hoặc nhiều short all-in cộng lại phải đủ
        # mức tố tối thiểu mới mở lại quyền tố.
        return self.current - previous >= self.min_raise

    def fraction_spend(self, seat, fraction):
        """
        Tổng tiền lấy thêm từ stack trong hành động này.
        Bao gồm phần theo và phần tố.
        """
        need = max(0, self.current - self.bets[seat])

        desired = math.ceil(self.chips[seat] * fraction)
        minimum = need + self.min_raise

        return min(
            self.chips[seat],
            max(desired, minimum),
        )

    # ========================================================
    # ENGINE: HÀNH ĐỘNG NO-LIMIT
    # ========================================================

    def poker_action(self, action):
        seat = self.turn

        if self.done:
            return

        if (
            seat in self.folded
            or self.chips[seat] <= 0
            or seat not in self.pending
        ):
            raise ValueError("Ghế này không có quyền hành động.")

        need = max(0, self.current - self.bets[seat])

        if action == "fold":
            self.folded.add(seat)
            self.pending.discard(seat)
            self.say(f"{self.name(seat)} FOLD.")

        elif action == "call":
            paid = poker_pay(self, seat, need)
            self.pending.discard(seat)

            self.last_action_level[seat] = self.current
            self.last_action_kind[seat] = (
                "check" if need == 0 else "call"
            )

            if need == 0:
                self.say(f"{self.name(seat)} CHECK.")
            elif self.chips[seat] == 0:
                self.say(
                    f"{self.name(seat)} CALL ALL-IN {paid:,} xu."
                )
            else:
                self.say(
                    f"{self.name(seat)} CALL {paid:,} xu."
                )

        elif action in (
            "bet25", "bet50", "bet75", "allin"
        ):
            if action == "allin":
                spend = self.chips[seat]
            else:
                fraction = {
                    "bet25": 0.25,
                    "bet50": 0.50,
                    "bet75": 0.75,
                }[action]

                spend = fraction_spend(
                    self, seat, fraction
                )

            target = self.bets[seat] + spend

            # All-in nhỏ hơn hoặc bằng số tiền cần theo.
            if target <= self.current:
                if spend != self.chips[seat]:
                    raise ValueError("Mức tiền này chưa đủ theo.")

                return poker_action(self, "call")

            if not can_raise(self, seat):
                raise ValueError(
                    "Chưa được mở lại quyền tố, "
                    "hoặc không còn đối thủ có thể theo thêm."
                )

            increase = target - self.current
            is_all_in = spend == self.chips[seat]

            if increase < self.min_raise and not is_all_in:
                raise ValueError(
                    f"Cần tố thêm tối thiểu {self.min_raise:,} xu."
                )

            full_raise = increase >= self.min_raise

            poker_pay(self, seat, spend)
            self.current = target

            if full_raise:
                self.min_raise = increase

            self.last_action_level[seat] = self.current
            self.last_action_kind[seat] = "raise"

            # Người chưa theo đủ mức mới phải hành động.
            # Quyền tố lại được kiểm tra riêng bằng can_raise().
            self.pending = {
                other
                for other in range(self.n)
                if other != seat
                and other not in self.folded
                and self.chips[other] > 0
                and self.bets[other] < self.current
            }

            label = "ALL-IN" if is_all_in else "BET/RAISE"

            self.say(
                f"{self.name(seat)} {label} "
                f"tới {target:,} xu trong vòng "
                f"(thêm {spend:,})."
            )

            if is_all_in and not full_raise:
                self.say(
                    "All-in thiếu mức tố tối thiểu: "
                    "không tự mở lại quyền tố."
                )

        else:
            raise ValueError("Hành động Poker không hợp lệ.")

        poker_progress(self, (seat + 1) % self.n)

    # ========================================================
    # ENGINE: CHUYỂN VÒNG / ALL-IN RUNOUT
    # ========================================================

    def poker_progress(self, start):
        while not self.done:
            live = live_seats(self)

            if len(live) == 1:
                settle_poker(self)
                return

            able = {
                seat
                for seat in live
                if self.chips[seat] > 0
            }

            self.pending.intersection_update(able)

            if len(able) <= 1:
                if able:
                    seat = next(iter(able))

                    # Khi chỉ một người còn tiền, chỉ cần theo
                    # phần cược thật sự có trên bàn.
                    actual_target = max(
                        self.bets[other]
                        for other in live
                    )

                    self.current = actual_target

                    if self.bets[seat] < actual_target:
                        self.pending = {seat}
                    else:
                        self.pending.clear()
                else:
                    self.pending.clear()

            if self.pending:
                for step in range(self.n):
                    seat = (start + step) % self.n

                    if seat in self.pending:
                        self.turn = seat
                        break

                street_name = (
                    "Pre-flop", "Flop", "Turn", "River"
                )[self.street]

                need = min(
                    self.chips[self.turn],
                    max(
                        0,
                        self.current - self.bets[self.turn],
                    ),
                )

                self.status = (
                    f"{street_name} • {self.name(self.turn)} "
                    f"• Cần theo {need:,} xu"
                )
                return

            if self.street == 3:
                settle_poker(self)
                return

            self.street += 1
            self.deck.pop()  # Burn card.

            count = 3 if self.street == 1 else 1

            for _ in range(count):
                self.board.append(self.deck.pop())

            self.bets = [0] * self.n
            self.current = 0
            self.min_raise = 20

            self.last_action_level = [None] * self.n
            self.last_action_kind = [None] * self.n

            self.pending = {
                seat
                for seat in live
                if self.chips[seat] > 0
            }

            name = ("", "FLOP", "TURN", "RIVER")[self.street]

            self.say(
                f"{name}: {cards_text(self.board)}"
            )

            start = (self.button + 1) % self.n

    # ========================================================
    # ENGINE: SIDE POT, HOÀN TIỀN, SHOWDOWN
    # ========================================================

    def settle_poker(self):
        live = live_seats(self)

        # Hoàn phần cược cao nhất không có ai theo.
        highest = max(self.contrib)
        top = [
            seat
            for seat, amount in enumerate(self.contrib)
            if amount == highest
        ]

        if highest > 0 and len(top) == 1:
            seat = top[0]

            second = max(
                (
                    amount
                    for other, amount in enumerate(self.contrib)
                    if other != seat
                ),
                default=0,
            )

            refund = highest - second

            if refund:
                self.contrib[seat] -= refund
                self.chips[seat] += refund

                self.say(
                    f"Hoàn {refund:,} xu chưa được theo "
                    f"cho {self.name(seat)}."
                )

        self.side_pots = []
        self.hand_values = {}
        winning_seats = set()

        if len(live) == 1:
            winner = live[0]
            amount = sum(self.contrib)

            self.chips[winner] += amount
            winning_seats.add(winner)

            self.side_pots.append(
                (amount, [winner])
            )

            self.say(
                f"{self.name(winner)} nhận {amount:,} xu "
                "vì mọi đối thủ đã fold."
            )

            self.showdown = False

        else:
            if len(self.board) != 5:
                raise RuntimeError(
                    "Showdown cần đủ năm lá bài chung."
                )

            self.showdown = True

            self.hand_values = {
                seat: poker_best(
                    self.hands[seat] + self.board
                )
                for seat in live
            }

            for seat in live:
                value = self.hand_values[seat]

                self.say(
                    f"{self.name(seat)}: "
                    f"{cards_text(self.hands[seat])} "
                    f"→ {poker_names[value[0]]} "
                    f"(so bộ: {value[1:]})"
                )

            levels = sorted({
                amount
                for amount in self.contrib
                if amount > 0
            })

            previous = 0

            for level in levels:
                contributors = [
                    seat
                    for seat in range(self.n)
                    if self.contrib[seat] >= level
                ]

                amount = (
                    level - previous
                ) * len(contributors)

                previous = level

                eligible = [
                    seat
                    for seat in contributors
                    if seat not in self.folded
                ]

                if not eligible:
                    raise RuntimeError(
                        "Side pot không có người đủ điều kiện nhận."
                    )

                best = max(
                    self.hand_values[seat]
                    for seat in eligible
                )

                winners = [
                    seat
                    for seat in eligible
                    if self.hand_values[seat] == best
                ]

                # Chip lẻ ưu tiên bên trái button.
                winners.sort(
                    key=lambda seat: (
                        seat - self.button - 1
                    ) % self.n
                )

                share, remainder = divmod(
                    amount, len(winners)
                )

                for index, seat in enumerate(winners):
                    self.chips[seat] += (
                        share + (index < remainder)
                    )

                winning_seats.update(winners)

                self.side_pots.append(
                    (amount, winners)
                )

                pot_name = (
                    "Main pot"
                    if len(self.side_pots) == 1
                    else f"Side pot {len(self.side_pots) - 1}"
                )

                self.say(
                    f"{pot_name}: {amount:,} xu → "
                    + ", ".join(
                        self.name(seat)
                        for seat in winners
                    )
                )

        # Bảo toàn toàn bộ chip của bàn.
        if sum(self.chips) != sum(self.starting_chips):
            raise RuntimeError(
                "Sai tổng chip sau chia pot. "
                "Dừng để tránh làm sai số dư."
            )

        self.hand_winners = sorted(winning_seats)

        # Giữ ranking cho tương thích nội bộ.
        # Không dùng nó để cộng điểm Poker Tournament.
        self.ranking = self.groups_by(
            self.chips, reverse=True
        )

        self.done = True
        self.turn = -1

        self.status = (
            "Kết thúc ván • "
            + ", ".join(
                self.name(seat)
                for seat in self.hand_winners
            )
            + " nhận pot"
        )

        self.say(self.status)

    # ========================================================
    # ENGINE: LOẠI BÀI VÀ AI
    # ========================================================

    def hand_label(self, seat):
        cards = self.hands[seat] + self.board

        if len(cards) >= 5:
            value = poker_best(cards)
            return poker_names[value[0]]

        # Pre-flop chỉ có hai lá.
        if self.hands[seat][0][0] == self.hands[seat][1][0]:
            return "Một đôi"

        return "Bài cao"

    def options(self):
        if self.kind != "Poker":
            return old_options(self)

        if self.done:
            return []

        seat = self.turn

        need = min(
            self.chips[seat],
            max(0, self.current - self.bets[seat]),
        )

        output = [
            ("Fold", "fold"),
            (
                "Check" if need == 0 else f"Call {need:,}",
                "call",
            ),
        ]

        if can_raise(self, seat):
            for label, action, fraction in (
                ("¼", "bet25", 0.25),
                ("½", "bet50", 0.50),
                ("¾", "bet75", 0.75),
            ):
                spend = fraction_spend(
                    self, seat, fraction
                )

                output.append(
                    (f"{label}: +{spend:,}", action)
                )

            output.append(
                (f"All-in {self.chips[seat]:,}", "allin")
            )

        elif self.chips[seat] <= max(
            0, self.current - self.bets[seat]
        ):
            output.append(
                (f"All-in {self.chips[seat]:,}", "allin")
            )

        return output

    def bot(self):
        if self.kind != "Poker":
            return old_bot(self)

        if self.done:
            return

        seat = self.turn
        opponents = len(live_seats(self)) - 1

        # Chỉ dùng bài riêng và bài chung.
        equity = equity_estimate(
            self.hands[seat],
            self.board,
            opponents,
            trials=16,
        )

        need = min(
            self.chips[seat],
            max(0, self.current - self.bets[seat]),
        )

        pot = sum(self.contrib)
        odds = need / (pot + need) if need else 0

        if need and equity + 0.025 < odds:
            if random.random() < 0.92:
                poker_action(self, "fold")
                return

        if can_raise(self, seat):
            if equity > 0.84:
                poker_action(self, "allin")
                return

            if equity > 0.69:
                poker_action(self, "bet50")
                return

            if equity > 0.53 and random.random() < 0.55:
                poker_action(self, "bet25")
                return

            # Bluff nhỏ, không dựa vào bài úp đối thủ.
            if (
                need == 0
                and opponents <= 2
                and random.random() < 0.035
            ):
                poker_action(self, "bet25")
                return

        poker_action(self, "call")

    def analyze(self):
        if self.kind != "Poker":
            return old_analyze(self)

        if self.done:
            lines = [
                "KẾT QUẢ POKER",
                "",
            ]

            for seat in sorted(
                range(self.n),
                key=lambda item: -self.chips[item],
            ):
                profit = (
                    self.chips[seat]
                    - self.starting_chips[seat]
                )

                if seat in self.folded:
                    category = "Đã fold — không công khai bài"
                elif self.showdown or self.ids[seat] == 0:
                    category = hand_label(self, seat)
                else:
                    category = "Thắng không cần lật bài"

                eliminated = (
                    " • HẾT XU"
                    if self.chips[seat] == 0
                    else ""
                )

                lines.append(
                    f"{self.name(seat)}: "
                    f"{self.chips[seat]:,} xu "
                    f"({profit:+,}){eliminated}\n"
                    f"  {category}"
                )

            lines.append("")

            for index, (amount, winners) in enumerate(
                self.side_pots
            ):
                label = (
                    "Main pot"
                    if index == 0
                    else f"Side pot {index}"
                )

                lines.append(
                    f"{label} {amount:,} → "
                    + ", ".join(
                        self.name(seat)
                        for seat in winners
                    )
                )

            return "\n".join(lines), []

        seat = self.human_seat()

        if seat is None:
            return (
                "Bạn đang quan sát.\n"
                "Các bot tiếp tục chơi bằng stack hiện có."
            ), []

        if seat in self.folded:
            return "Bạn đã fold, đang chờ kết thúc ván.", []

        category = hand_label(self, seat)

        opponents = len(live_seats(self)) - 1

        equity = equity_estimate(
            self.hands[seat],
            self.board,
            opponents,
            trials=60,
        )

        return (
            f"Bộ tốt nhất hiện tại: {category}\n"
            f"Stack: {self.chips[seat]:,} xu\n"
            f"Đã góp vào pot: {self.contrib[seat]:,} xu\n"
            f"Equity tham khảo: {equity:.1%}\n\n"
            "¼ / ½ / ¾ tính trên stack còn lại. "
            "Game tự nâng lên mức tố tối thiểu khi cần.\n\n"
            "Bộ bài có thể sử dụng bài chung trên bàn. "
            "Equity giả định bài đối thủ ngẫu nhiên, "
            "chưa phân tích đầy đủ range hoặc từng side pot."
        ), []

    def summary(self):
        if self.kind != "Poker":
            return old_summary(self)

        return (
            f"POT {sum(self.contrib):,} xu • NO-LIMIT\n"
            f"SB: {self.name(self.small_blind)} "
            f"• BB: {self.name(self.big_blind)}"
        )

    # ========================================================
    # TOURNAMENT POKER: STACK THẬT, TỐI ĐA 3 VÁN/BÀN
    # ========================================================

    class PokerTournament:
        def __init__(self, kind="Poker"):
            self.kind = "Poker"
            self.entrants = list(range(32))

            seeds = list(range(32))
            random.shuffle(seeds)

            self.seed = {
                pid: index
                for index, pid in enumerate(seeds)
            }

            self.bank = {
                pid: 1000
                for pid in range(32)
            }

            # Tổng số lần tái mua tự động trong cả giải.
            self.rebuys = {
                pid: 0
                for pid in range(32)
            }

            self.history = []
            self.out_stage = {}
            self.qualifiers = []
            self.complete = False
            self.final_order = []
            self.champion = None

            self.start_stage()

        @property
        def current_group(self):
            return self.groups[self.group_index]

        @property
        def label(self):
            if self.stage_size <= 4:
                return "FINAL TABLE"
            return f"VÒNG {self.stage_size}"

        def start_stage(self):
            self.stage_size = len(self.entrants)

            ids = list(self.entrants)
            random.shuffle(ids)

            # Chia cân bằng, tránh bàn một người.
            table_count = max(1, math.ceil(len(ids) / 4))

            self.groups = [
                ids[index::table_count]
                for index in range(table_count)
            ]

            self.groups.sort(
                key=lambda group: 0 not in group
            )

            self.group_index = 0
            self.qualifiers = []
            self.reset_match()

        def reset_match(self):
            self.hand_index = 0
            self.previous_button = None

            self.points = {
                pid: self.bank[pid]
                for pid in self.current_group
            }

            self.wins = {
                pid: 0
                for pid in self.current_group
            }

        def match_order(self):
            return sorted(
                self.current_group,
                key=lambda pid: (
                    -self.bank[pid],
                    self.seed[pid],
                ),
            )

        def record_hand(self, engine):
            for seat, pid in enumerate(engine.ids):
                previous_stack = self.bank[pid]
                ending_stack = engine.chips[seat]

                self.bank[pid] = ending_stack
                self.points[pid] = ending_stack

                if previous_stack > 0 and ending_stack == 0:
                    if (
                        self.rebuy_allowed()
                        and self.hand_index < 2
                    ):
                        engine.say(
                            f"🔄 {player_name(pid)} hết xu. "
                            "Ván tiếp theo tại vòng 32 "
                            "sẽ được cấp lại 1.000 xu."
                        )
                    else:
                        engine.say(
                            f"☠ {player_name(pid)} hết xu "
                            f"tại {self.label} — bị loại, "
                            "không được cấp lại tiền."
                        )

        def rebuy_allowed(self):
            """
            Chỉ vòng đầu 32 người được cấp lại tiền.
            Các vòng sau, kể cả vòng có số người không chuẩn,
            đều không được cấp lại.
            """
            return self.stage_size == 32

        def match_finished(self):
            # Đã chơi xong ván 3 thì chốt bàn.
            if self.hand_index >= 2:
                return True

            # Vòng 32 luôn chơi đủ 3 ván.
            if self.rebuy_allowed():
                return False

            # Từ vòng 16 trở đi:
            # còn một người có xu thì kết thúc bàn sớm.
            alive = [
                pid
                for pid in self.current_group
                if self.bank[pid] > 0
            ]

            return len(alive) <= 1

        def advance(self):
            if self.complete:
                return

            if not self.match_finished():
                self.hand_index += 1
                return

            order = self.match_order()

            self.history.append({
                "stage": self.stage_size,
                "order": list(order),
                "points": {
                    pid: self.bank[pid]
                    for pid in self.current_group
                },
                "hands": self.hand_index + 1,
            })

            if self.stage_size <= 4:
                self.champion = order[0]

                older = sorted(
                    self.out_stage,
                    key=lambda pid: (
                        self.out_stage[pid],
                        -self.bank[pid],
                        self.seed[pid],
                    ),
                )

                self.final_order = order + older
                self.complete = True
                return

            positive = [
                pid
                for pid in order
                if self.bank[pid] > 0
            ]

            advancing = positive[:2]
            self.qualifiers.extend(advancing)

            for pid in order:
                if pid not in advancing:
                    self.out_stage[pid] = self.stage_size

            self.group_index += 1

            if self.group_index < len(self.groups):
                self.reset_match()
                return

            self.entrants = list(self.qualifiers)

            if len(self.entrants) == 1:
                self.champion = self.entrants[0]
                self.complete = True

                self.final_order = (
                    self.entrants
                    + sorted(
                        self.out_stage,
                        key=lambda pid: (
                            self.out_stage[pid],
                            -self.bank[pid],
                            self.seed[pid],
                        ),
                    )
                )
                return

            self.start_stage()

        def report(self):
            lines = [
                "POKER TOURNAMENT — STACK THẬT",

                        f"{player_name(pid)} "

                    )

            return "\n".join(lines)

    def next_button_pid(roster, alive, previous):
        if previous is None:
            return next(pid for pid in roster if pid in alive)

        start = roster.index(previous)

        for step in range(1, len(roster) + 1):
            pid = roster[(start + step) % len(roster)]

            if pid in alive:
                return pid

        raise RuntimeError("Không còn người để đặt button.")

    def start_poker_hand(self):
        self.cancel_timer()
        self.selected.clear()
        self.bubbles.clear()
        self.accounted = False

        t = self.tournament

        if isinstance(t, PokerTournament):
            if t.complete:
                self.render()
                self.show_standings()
                return

            roster = list(t.current_group)

            # Tái mua trước ván 2 và ván 3.
            #
            # Không làm trong record_hand(), vì sau ván 3
            # phải dùng số xu thật để xét đi tiếp.
            # Chỉ vòng 32 được tái mua trước ván 2 và 3.
            if (
                t.rebuy_allowed()
                and t.hand_index in (1, 2)
            ):
                for pid in roster:
                    if t.bank[pid] == 0:
                        t.bank[pid] = 1000
                        t.points[pid] = 1000
                        t.rebuys[pid] += 1

                        self.write(
                            f"🔄 VÒNG 32: {player_name(pid)} "
                            f"được cấp lại 1.000 xu để chơi "
                            f"ván {t.hand_index + 1}/3."
                        )

            bank = t.bank
            previous_button = t.previous_button

            label = (
                f"{t.label} • "
                f"Bàn {t.group_index + 1}/{len(t.groups)} "
                f"• Ván {t.hand_index + 1}/3"
            )

        else:
            if self.quick_hand == 0:
                count = int(self.seats.get())

                self._poker_quick = {
                    "roster": list(range(count)),
                    "bank": {
                        pid: 1000
                        for pid in range(count)
                    },
                    "places": {},
                    "eliminated": [],
                    "previous_button": None,
                    "complete": False,
                }

            context = self._poker_quick
            roster = list(context["roster"])
            bank = context["bank"]
            previous_button = context["previous_button"]

            label = (
                f"Poker ván nhanh #{self.quick_hand + 1}"
                " • Giữ nguyên stack"
            )

        alive = [
            pid
            for pid in roster
            if bank[pid] > 0
        ]

        if len(alive) < 2:
            raise RuntimeError(
                "Không được mở ván mới khi còn dưới hai người."
            )

        button_pid = next_button_pid(
            roster, alive, previous_button
        )

        if isinstance(t, PokerTournament):
            t.previous_button = button_pid
        else:
            self._poker_quick["previous_button"] = button_pid

        self.silent = 0 not in alive

        self.write("\n" + "━" * 25)
        self.write(label)

        self.game = Engine(
            "Poker",
            alive,
            hand_number=alive.index(button_pid),
            logger=self.engine_log,
            starting_stacks=[
                bank[pid] for pid in alive
            ],
        )

        self.set_analysis(
            "No-limit Poker.\n"
            "Stack được giữ sang ván sau.\n"
            "SB/BB được ghi trên bàn và tại ghế."
        )

        self.render()
        self.pump()

    def start_hand(self):
        poker_tournament = isinstance(
            self.tournament, PokerTournament
        )

        poker_quick = (
            self.tournament is None
            and self.kind.get() == "Poker"

        if poker_tournament or poker_quick:
            return start_poker_hand(self)

        return old_start_hand(self)

    def record_quick_result(self, game):
        context = self._poker_quick
        bank = context["bank"]

        for seat, pid in enumerate(game.ids):
            bank[pid] = game.chips[seat]

        busted = [
            seat
            for seat in range(game.n)
            if game.chips[seat] == 0
            and game.ids[seat] not in context["places"]
        ]

        # Cùng bị loại trong một ván:
        # stack đầu ván nhỏ hơn nhận vị trí thấp hơn.
        # Bằng stack: dùng thứ tự ghế làm tiêu chí cuối.
        busted.sort(
            key=lambda seat: (
                game.starting_chips[seat],
                -seat,
            )
        )

        total = len(context["roster"])

        for seat in busted:
            pid = game.ids[seat]

            place = total - len(context["eliminated"])

            context["places"][pid] = place
            context["eliminated"].append(pid)

            self.write(
                f"☠ {player_name(pid)} hết xu — "
                f"kết thúc hạng {place}."
            )

        survivors = [
            pid
            for pid in context["roster"]
            if bank[pid] > 0
        ]

        if len(survivors) == 1:
            winner = survivors[0]
            context["places"][winner] = 1
            context["complete"] = True

            self.write(
                f"🏆 {player_name(winner)} thắng phiên Poker "
                f"với {bank[winner]:,} xu!"
            )

    def pump(self):
        game = self.game

        if game is None or game.kind != "Poker":
            return old_pump(self)

        if self.paused:
            self.render()
            return

        if game.done:
            if not self.accounted:
                self.accounted = True

                if isinstance(
                    self.tournament, PokerTournament
                ):
                    self.tournament.record_hand(game)
                else:
                    record_quick_result(self, game)

                text, _ = game.analyze()
                self.set_analysis(text)
                self.write(game.status)

                if self.sound.get() and not self.silent:
                    self.bell()

            self.render()

            quick_complete = (
                self.tournament is None
                and self._poker_quick["complete"]
            )

            if self.silent and not quick_complete:
                self.schedule(self.next_hand, 250)

            return

        self.render()

        if self.silent:
            self.schedule(self.simulate_table_hand, 30)
            return

        if game.ids[game.turn] != 0:
            delay = {
                "Chậm": 1400,
                "Vừa": 800,
                "Nhanh": 220,
            }[self.speed.get()]

            self.schedule(self.bot_step, delay)

    def next_hand(self):
        game = self.game

        if game is None or game.kind != "Poker":
            return old_next_hand(self)

        if self.paused or not game.done:
            return

        self.cancel_timer()

        if not self.accounted:
            self.pump()

        t = self.tournament

        if isinstance(t, PokerTournament):
            if t.complete:
                self.show_standings()
                return

            if t.match_finished():
                self.write(
                    "Chốt bàn Poker: "
                    + " | ".join(
                        f"{player_name(pid)} "
                        f"{t.bank[pid]:,} xu"
                        for pid in t.match_order()
                    )
                )

            t.advance()

            if t.complete:
                self.render()
                self.show_standings()
                return
        else:
            if self._poker_quick["complete"]:
                self.show_standings()
                return

            self.quick_hand += 1

        start_poker_hand(self)

    def seat_details(self, seat):
        game = self.game

        if game.kind != "Poker":
            return old_seat_details(self, seat)

        tags = []

        if seat == game.small_blind:
            tags.append("SB")

        if seat == game.big_blind:
            tags.append("BB")

        if seat in game.folded:
            state = "FOLD"

        elif game.chips[seat] == 0:
            state = (
                "HẾT XU"
                if game.done
                else "ALL-IN"
            )

        else:
            state = ""

        # Chỉ hiện dạng bài người thật, hoặc bài đã showdown.
        reveal_type = (
            game.ids[seat] == 0
            or (
                game.done
                and game.showdown
                and seat not in game.folded
            )
        )

        category = (
            hand_label(game, seat)
            if reveal_type
            else ""
        )

        if game.ids[seat] == 0:
            details = [
                f"{game.chips[seat]:,} xu",
                f"Cược vòng {game.bets[seat]:,}",
                "/".join(tags),
                state,
                category,
            ]

            return " • ".join(
                item for item in details if item
            )

        lower = " • ".join(
            item
            for item in (state, category)
            if item
        )

        return (
            f"{game.chips[seat]:,} xu"
            + (f"\n{lower}" if lower else "")
        )

class BauCuaEngine:
    def __init__(self, ids, stacks, logger=None):
        if not ids or len(ids) != len(stacks):
            raise ValueError("Danh sách người chơi/stack không hợp lệ.")

        if any(type(value) is not int or value <= 0 for value in stacks):
            raise ValueError("Stack phải là số nguyên dương.")

        self.kind = "Bầu cua"
        self.ids = list(ids)
        self.n = len(ids)
        self.logger = logger or (lambda text: None)

        self.starting_chips = list(stacks)
        self.chips = list(stacks)

        # Giữ các trường giao tiếp mà ứng dụng đang sử dụng.
        self.hands = [[] for _ in ids]
        self.board = []
        self.table = []
        self.ranking = []

        self.wagers = [[0] * 6 for _ in ids]
        self.locked = set()
        self.selected_face = 0

        self.dice = []
        self.payouts = [0] * self.n
        self.turn = 0
        self.done = False
        self.status = f"{self.name(0)} chọn cửa và đặt cược."

    def name(self, seat):
        return player_name(self.ids[seat])

    def say(self, text):
        self.logger(text)

    def human_seat(self):
        return self.ids.index(0) if 0 in self.ids else None

    def available(self, seat):
        if seat in self.locked:
            return self.chips[seat]
        return self.chips[seat] - sum(self.wagers[seat])

    def options(self):
        if self.done:
            return []

        return [
            ("+10", "add:10"),
            ("+50", "add:50"),
            ("+100", "add:100"),
            ("¼ còn lại", "fraction:1"),
            ("½ còn lại", "fraction:2"),
            ("¾ còn lại", "fraction:3"),
            ("All-in", "allin"),
            ("Xóa cược", "clear"),
            ("CHỐT CƯỢC", "confirm"),
        ]

    def action(self, action, selected=()):
        if self.done:
            return

        seat = self.turn

        if seat in self.locked:
            raise ValueError("Người chơi này đã chốt cược.")

        if action.startswith("face:"):
            face = int(action.split(":")[1])

            if not 0 <= face < 6:
                raise ValueError("Cửa cược không hợp lệ.")

            self.selected_face = face
            return

        if action == "clear":
            self.wagers[seat] = [0] * 6
            return

        if action == "confirm":
            total = sum(self.wagers[seat])
            minimum = min(10, self.starting_chips[seat])

            if not minimum <= total <= self.chips[seat]:
                raise ValueError(
                    f"Tổng cược phải từ {minimum:,} "
                    f"đến {self.chips[seat]:,} xu."
                )

            self.chips[seat] -= total
            self.locked.add(seat)

            description = ", ".join(
                f"{BC_FACES[face]} {amount:,}"
                for face, amount in enumerate(self.wagers[seat])
                if amount
            )

            self.say(f"{self.name(seat)} chốt: {description}.")

            if len(self.locked) == self.n:
                self.settle()
            else:
                self.turn = next(
                    other
                    for other in range(self.n)
                    if other not in self.locked
                )
                self.selected_face = 0
                self.status = (
                    f"{self.name(self.turn)} chọn cửa và đặt cược."
                )
            return

        available = self.available(seat)

        if action.startswith("add:"):
            amount = int(action.split(":")[1])

        elif action.startswith("fraction:"):
            numerator = int(action.split(":")[1])

            if numerator not in (1, 2, 3):
                raise ValueError("Tỷ lệ cược không hợp lệ.")

            amount = (available * numerator + 3) // 4

        elif action == "allin":
            amount = available

        else:
            raise ValueError("Hành động Bầu cua không hợp lệ.")

        if amount <= 0:
            raise ValueError("Không còn tiền chưa phân bổ.")

        if amount > available:
            raise ValueError(
                f"Chỉ còn {available:,} xu chưa phân bổ. "
                "Hãy giảm cược hoặc dùng All-in."
            )

        self.wagers[seat][self.selected_face] += amount

    def settle(self):
        if self.done:
            return

        if len(self.locked) != self.n:
            raise RuntimeError("Chưa đủ người chốt cược.")

        # Không sinh kết quả trước khi tất cả đã chốt.
        rng = random.SystemRandom()
        self.dice = [rng.randrange(6) for _ in range(3)]
        counts = Counter(self.dice)

        self.say(
            "🎲 KẾT QUẢ: "
            + " • ".join(BC_FACES[face] for face in self.dice)
        )

        for seat in range(self.n):
            payout = sum(
                amount * (counts[face] + 1)
                for face, amount in enumerate(self.wagers[seat])
                if counts[face] > 0
            )

            self.payouts[seat] = payout
            self.chips[seat] += payout

            expected = (
                self.starting_chips[seat]
                - sum(self.wagers[seat])
                + payout
            )

            if self.chips[seat] != expected or expected < 0:
                raise RuntimeError("Sai số dư sau trả thưởng.")

            profit = self.chips[seat] - self.starting_chips[seat]

            self.say(
                f"{self.name(seat)} nhận {payout:,} xu; "
                f"lãi/lỗ {profit:+,}; "
                f"còn {self.chips[seat]:,} xu."
            )

        self.done = True
        self.turn = -1
        self.status = (
            "Kết quả: "
            + " • ".join(BC_FACES[face] for face in self.dice)
        )

    def bot(self):
        if self.done:
            return

        seat = self.turn
        stack = self.chips[seat]

        # Cược ngẫu nhiên 25–100% stack, chia 1–3 cửa.
        # Không dùng lịch sử để giả vờ dự đoán xúc xắc.
        numerator = random.choice((1, 2, 3, 4))
        budget = min(
            stack,
            max(min(10, stack), stack * numerator // 4),
        )

        number = min(random.randint(1, 3), budget)
        faces = random.sample(range(6), number)

        share, extra = divmod(budget, number)

        self.wagers[seat] = [0] * 6

        for index, face in enumerate(faces):
            self.wagers[seat][face] = share + (index < extra)

        self.action("confirm")

    def analyze(self):
        if not self.done:
            return (
                "BẦU CUA\n\n"
                "Chọn cửa trên bàn rồi thêm tiền cược.\n"
                "Có thể chia tiền vào nhiều cửa.\n"
                "Bấm CHỐT CƯỢC để xác nhận.\n\n"
                "Các cửa có xác suất như nhau. "
                "Kết quả cũ không dự đoán được lượt tiếp theo.\n"
                "Đặt nhiều cửa không bảo đảm có lãi."
            ), []

        lines = [self.status, ""]

        for seat in sorted(
            range(self.n),
            key=lambda item: -self.chips[item],
        ):
            profit = self.chips[seat] - self.starting_chips[seat]

            lines.append(
                f"{self.name(seat)}: {self.chips[seat]:,} xu "
                f"({profit:+,})"
            )
            lines.append(
                f"  Cược {sum(self.wagers[seat]):,}"
                f" • Nhận lại {self.payouts[seat]:,}"
            )

        return "\n".join(lines), []

class BauCuaSession:
    def __init__(self, tournament=False, seats=4):
        self.kind = "Bầu cua"
        self.is_tournament = tournament

        count = 32 if tournament else seats

        self.bank = {pid: 1000 for pid in range(count)}
        self.rebuys = {pid: 0 for pid in range(count)}
        self.before = dict(self.bank)

        draw = list(self.bank)
        random.shuffle(draw)
        self.seed = {
            pid: index for index, pid in enumerate(draw)
        }

        self.entrants = list(self.bank)
        self.round_index = 0
        self.complete = False
        self.champions = []
        self.places = {}
        self.states = {pid: "Đang tham gia" for pid in self.bank}
        self.history = []

        if tournament:
            self.start_stage()
        else:
            self.stage_size = count
            self.groups = [list(self.bank)]
            self.group_index = 0
            self.hand_index = 0

    @property
    def current_group(self):
        return self.groups[self.group_index]

    @property
    def label(self):
        if not self.is_tournament:
            return "VÁN NHANH"

        if self.stage_size <= 4:
            return "FINAL TABLE"

        return f"VÒNG {self.stage_size}"

    def start_stage(self):
        self.stage_size = len(self.entrants)
        ids = list(self.entrants)
        random.shuffle(ids)

        table_count = max(1, math.ceil(len(ids) / 4))

        self.groups = [
            ids[index::table_count]
            for index in range(table_count)
        ]
        self.groups.sort(key=lambda group: 0 not in group)

        self.group_index = 0
        self.hand_index = 0
        self.qualifiers = []

    def prepare(self):
        if self.complete:
            raise RuntimeError("Phiên đã kết thúc.")

        messages = []

        # Chỉ vòng đầu của giải được tái mua.
        if (
            self.is_tournament
            and self.round_index == 0
            and self.hand_index in (1, 2)
        ):
            for pid in self.current_group:
                if self.bank[pid] == 0:
                    self.bank[pid] = 1000
                    self.rebuys[pid] += 1
                    self.states[pid] = "Được cấp lại tiền"
                    messages.append(
                        f"🔄 {player_name(pid)} nhận lại "
                        "1.000 xu ở vòng 32."
                    )

        ids = [
            pid
            for pid in self.current_group
            if self.bank[pid] > 0
        ]

        if not ids:
            raise RuntimeError("Không có người đủ tiền để mở lượt.")

        return ids, messages

    def record_hand(self, game):
        if not game.done:
            raise ValueError("Chưa thể ghi kết quả lượt đang chơi.")

        for seat, pid in enumerate(game.ids):
            self.before[pid] = game.starting_chips[seat]
            self.bank[pid] = game.chips[seat]

            if self.bank[pid] > 0:
                self.states[pid] = "Còn xu"
            elif (
                self.is_tournament
                and self.round_index == 0
                and self.hand_index < 2
            ):
                self.states[pid] = "Chờ cấp lại 1.000 xu"
            else:
                self.states[pid] = "Đã hết xu"

        self.history.append(
            f"{self.label} • Bàn {self.group_index + 1}"
            f" • Lượt {self.hand_index + 1}: "
            + " / ".join(BC_FACES[face] for face in game.dice)
            + "\n  "
            + " | ".join(
                f"{player_name(pid)}: {self.bank[pid]:,}"
                for pid in game.ids
            )
        )

        if self.is_tournament:
            return

        # Xếp hạng ván nhanh theo thứ tự bị loại.
        survivors = [
            pid for pid in self.bank if self.bank[pid] > 0
        ]

        busted = [
            pid for pid in game.ids
            if self.bank[pid] == 0 and pid not in self.places
        ]

        busted.sort(key=lambda pid: -self.before[pid])

        position = len(survivors) + 1

        for _, group in itertools.groupby(
            busted,
            key=lambda pid: self.before[pid],
        ):
            tied = list(group)

            for pid in tied:
                self.places[pid] = position

            position += len(tied)

        if len(survivors) <= 1:
            self.complete = True

            if survivors:
                self.places[survivors[0]] = 1

            self.champions = [
                pid for pid, rank in self.places.items()
                if rank == 1
            ]

    def match_order(self):
        return sorted(
            self.current_group,
            key=lambda pid: (-self.bank[pid], self.seed[pid]),
        )

    def match_finished(self):
        if self.hand_index >= 2:
            return True

        if self.round_index == 0:
            return False

        return sum(
            self.bank[pid] > 0
            for pid in self.current_group
        ) <= 1

    def advance(self):
        if self.complete:
            return

        if not self.is_tournament:
            self.hand_index += 1
            return

        if not self.match_finished():
            self.hand_index += 1
            return

        order = self.match_order()
        positive = [pid for pid in order if self.bank[pid] > 0]

        if self.stage_size <= 4:
            if positive:
                winner = positive[0]
            else:
                # Final cùng hết tiền:
                # xét số dư trước lượt cuối rồi hạt giống.
                winner = min(
                    self.current_group,
                    key=lambda pid: (
                        -self.before[pid],
                        self.seed[pid],
                    ),
                )

            self.champions = [winner]
            self.complete = True

            for pid in self.current_group:
                self.states[pid] = (
                    "VÔ ĐỊCH" if pid == winner else "Dừng ở Final"
                )
            return

        advancing = positive[:2]
        self.qualifiers.extend(advancing)

        for pid in self.current_group:
            self.states[pid] = (
                "Đi tiếp"
                if pid in advancing
                else f"Dừng ở vòng {self.stage_size}"
            )

        self.group_index += 1

        if self.group_index < len(self.groups):
            self.hand_index = 0
            return

        self.entrants = list(self.qualifiers)

        # Không tạo vòng/bàn rỗng.
        if len(self.entrants) <= 1:
            self.complete = True
            self.champions = list(self.entrants)

            if self.champions:
                self.states[self.champions[0]] = "VÔ ĐỊCH"
            return

        self.round_index += 1
        self.start_stage()

    def report(self):
        lines = [
            "BẦU CUA — " + (
                "TOURNAMENT" if self.is_tournament else "VÁN NHANH"
            ),
            "Số dư được cập nhật sau mỗi lượt lắc.",
            "",
        ]

        if self.complete:
            if self.champions:
                lines.append(
                    "🏆 KẾT QUẢ: "
                    + ", ".join(
                        player_name(pid) for pid in self.champions
                    )
                )
            else:
                lines.append(
                    "Giải kết thúc: không còn người đủ tiền đi tiếp."
                )
        else:
            lines.append(
                f"{self.label} • Bàn {self.group_index + 1}"
                f" • Lượt {self.hand_index + 1}"
            )

        lines.append("")

        if self.is_tournament:
            order = sorted(
                self.bank,
                key=lambda pid: (-self.bank[pid], self.seed[pid]),
            )

            lines.append(
                "Danh sách theo số dư; trạng thái cho biết vòng tiến tới."
            )
        else:
            order = sorted(
                self.bank,
                key=lambda pid: (
                    self.places.get(pid, 0),
                    -self.bank[pid],
                    pid,
                ),
            )

        for pid in order:
            if self.is_tournament:
                status = self.states[pid]
            elif pid in self.places:
                status = f"Hạng {self.places[pid]}"
            else:
                status = "Đang chơi"

            marker = " ← BẠN" if pid == 0 else ""

            lines.append(
                f"{player_name(pid):<14}"
                f" | {self.bank[pid]:>8,} xu"
                f" | {status}"
                f" | Cấp lại {self.rebuys[pid]} lần"
                f"{marker}"
            )

        lines.append("\nLỊCH SỬ — TỐI ĐA 60 LƯỢT GẦN NHẤT")
        lines.extend(self.history[-60:])

        return "\n".join(lines)

def install_baucua(namespace):
    if namespace.get("_BAUCUA_INSTALLED"):
        return

    namespace["_BAUCUA_INSTALLED"] = True

    Lounge = namespace["CardLounge"]
    previous_factory = namespace["Tournament"]

    old_start = Lounge.start_hand
    old_next = Lounge.next_hand
    old_pump = Lounge.pump
    old_render = Lounge.render
    old_standings = Lounge.show_standings
    old_settings = Lounge.settings_changed
    old_rules = Lounge.show_rules

    def factory(kind):
        if kind == "Bầu cua":
            return BauCuaSession(tournament=True)
        return previous_factory(kind)

    def settings_changed(self, event=None):
        old_settings(self, event)

        if (
            self.kind.get() == "Bầu cua"
            and self.play_mode.get() == "Ván nhanh"
        ):
            self.seat_box.configure(state="normal")

    def start_hand(self):
        is_tournament = isinstance(
            self.tournament, BauCuaSession
        )

        is_quick = (
            self.tournament is None
            and self.kind.get() == "Bầu cua"
        )

        if not (is_tournament or is_quick):
            return old_start(self)

        self.cancel_timer()
        self.selected.clear()
        self.bubbles.clear()
        self.accounted = False

        if is_tournament:
            session = self.tournament
        else:
            if self.quick_hand == 0:
                self._bc_quick = BauCuaSession(
                    tournament=False,
                    seats=int(self.seats.get()),
                )

            session = self._bc_quick

        self._bc_session = session

        if session.complete:
            self.render()
            self.show_standings()
            return

        ids, messages = session.prepare()

        for text in messages:
            self.write(text)

        self.silent = 0 not in ids

        self.game = BauCuaEngine(
            ids,
            [session.bank[pid] for pid in ids],
            logger=self.engine_log,
        )

        self.write(
            f"\n🎲 BẦU CUA • {session.label}"
            f" • Bàn {session.group_index + 1}"
            f" • Lượt {session.hand_index + 1}"
        )

        self.set_analysis(self.game.analyze()[0])
        self.pump()

    def pump(self):
        game = self.game

        if game is None or game.kind != "Bầu cua":
            return old_pump(self)

        if self.paused:
            self.render()
            return

        session = self._bc_session

        if game.done:
            if not self.accounted:
                session.record_hand(game)
                self.accounted = True

                self.set_analysis(game.analyze()[0])
                self.write(game.status)

                if self.sound.get() and not self.silent:
                    self.bell()

            self.render()

            if self.silent and not session.complete:
                self.schedule(self.next_hand, 500)

            return

        self.render()

        if self.silent:
            self.schedule(self.simulate_table_hand, 40)

        elif game.ids[game.turn] != 0:
            delay = {
                "Chậm": 1200,
                "Vừa": 650,
                "Nhanh": 180,
            }[self.speed.get()]

            self.schedule(self.bot_step, delay)

    def next_hand(self):
        game = self.game

        if game is None or game.kind != "Bầu cua":
            return old_next(self)

        if self.paused or not game.done:
            return

        if not self.accounted:
            self.pump()

        self.cancel_timer()
        session = self._bc_session

        if session.complete:
            self.show_standings()
            return

        session.advance()

        if session.complete:
            self.write("🏁 Phiên Bầu cua đã kết thúc.")
            self.render()
            self.show_standings()
            return

        if not session.is_tournament:
            self.quick_hand = session.hand_index

        self.start_hand()

    def show_standings(self):
        if isinstance(self.tournament, BauCuaSession):
            session = self.tournament
        elif (
            self.game is not None
            and self.game.kind == "Bầu cua"
            and self.tournament is None
        ):
            session = self._bc_session
        else:
            return old_standings(self)

        self.text_window(
            "Bầu cua — Kết quả và số dư",
            session.report(),
            width=1000,
            height=700,
        )

    def show_rules(self):
        if self.kind.get() == "Bầu cua":
            self.text_window(
                "Luật Bầu cua",
                BC_RULES.strip(),
                width=850,
                height=700,
            )
            return

        return old_rules(self)

    def custom_bet(self):
        from tkinter import simpledialog

        game = self.game

        if (
            self.paused
            or game is None
            or game.kind != "Bầu cua"
            or game.done
            or game.ids[game.turn] != 0
        ):
            return

        available = game.available(game.turn)

        if available <= 0:
            messagebox.showinfo(
                "Đặt cược",
                "Bạn đã phân bổ hết tiền. "
                "Có thể xóa cược để đặt lại.",
            )
            return

        amount = simpledialog.askinteger(
            "Nhập tiền cược",
            f"Cửa {BC_FACES[game.selected_face]}\n"
            f"Nhập từ 1 đến {available:,} xu:",
            parent=self,
            minvalue=1,
            maxvalue=available,
        )

        if (
            amount is not None
            and self.game is game
            and not self.paused
            and not game.done
        ):
            self.human_action(f"add:{amount}")

    def render(self):
        game = self.game

        if game is None or game.kind != "Bầu cua":
            return old_render(self)

        canvas = self.canvas
        canvas.delete("all")
        self.card_hitboxes = []

        # Bầu cua dùng các vùng chọn cửa, không chọn lá bài.
        canvas.unbind("<Button-1>")

        width = max(600, canvas.winfo_width())
        height = max(420, canvas.winfo_height())

        background, felt, edge, _ = THEMES[self.theme.get()]
        canvas.configure(bg=background)

        session = self._bc_session

        label = (
            "ĐÃ KẾT THÚC"
            if session.complete
            else (
                f"{session.label}"
                f" • Bàn {session.group_index + 1}"
                f" • Lượt {session.hand_index + 1}"
                + ("/3" if session.is_tournament else "")
            )
        )

        self.banner.configure(text=label)

        canvas.create_text(
            width / 2, 23,
            text=f"BẦU CUA • {label}",
            fill="#ffe3a0",
            font=("Segoe UI", 14, "bold"),
        )

        result = (
            "  |  ".join(BC_FACES[face] for face in game.dice)
            if game.done else "XÚC XẮC CHƯA LẮC"
        )

        canvas.create_text(
            width / 2, 64,
            text=result,
            fill="#ffffff",
            font=("Segoe UI", 21, "bold"),
        )

        human = game.human_seat()

        editable = (
            not self.paused
            and not game.done
            and human is not None
            and game.turn == human
        )

        # Khu cửa cược chiếm bên trái; số dư ở bên phải.
        board_left = 18
        board_right = width * 0.66
        board_top = 108
        board_bottom = height - 94

        gap = 9
        cell_width = (board_right - board_left - 2 * gap) / 3
        cell_height = (board_bottom - board_top - gap) / 2

        for face, name in enumerate(BC_FACES):
            row, column = divmod(face, 3)

            x = board_left + column * (cell_width + gap)
            y = board_top + row * (cell_height + gap)

            tag = f"bc-face-{face}"
            chosen = editable and game.selected_face == face

            canvas.create_rectangle(
                x, y,
                x + cell_width, y + cell_height,
                fill="#394b38" if chosen else felt,
                outline="#ffe18d" if chosen else edge,
                width=3 if chosen else 1,
                tags=(tag,),
            )

            canvas.create_text(
                x + cell_width / 2,
                y + cell_height * 0.28,
                text=name.upper(),
                fill="#fff0c8",
                font=("Segoe UI", 16, "bold"),
                tags=(tag,),
            )

            own = (
                game.wagers[human][face]
                if human is not None else 0
            )

            detail = (
                f"Bạn cược: {own:,}"
                if human is not None else "Bàn mô phỏng"
            )

            if game.done:
                detail += f"\nXuất hiện: {game.dice.count(face)} lần"

            canvas.create_text(
                x + cell_width / 2,
                y + cell_height * 0.65,
                text=detail,
                fill="#e3eaf4",
                font=("Segoe UI", 10),
                tags=(tag,),
            )

            if editable:
                canvas.tag_bind(
                    tag,
                    "<Button-1>",
                    lambda event, value=face:
                        self.human_action(f"face:{value}"),

def install_tienlen_lung(namespace):
    if namespace.get("_TIENLEN_LUNG_INSTALLED"):
        return

    Engine = namespace["Engine"]
    Lounge = namespace["CardLounge"]

    old_start_tl = Engine.start_tl
    old_analyze = Engine.analyze
    old_seat_details = Lounge.seat_details

    def start_tl(self):
        # Các ghế bị lủng trong ván hiện tại.
        self.lung_seats = set()

        # Chỉ kiểm tra lủng một lần, lúc có người về nhất.
        self.lung_checked = False

        old_start_tl(self)

    def tl_active(self):
        """Chỉ những người chưa kết thúc mới được đánh tiếp."""
        lung = getattr(self, "lung_seats", set())

        return [
            seat
            for seat in range(self.n)
            if seat not in self.finished
            and seat not in self.thoi_two
            and seat not in lung
        ]

    def tl_finish_if_needed(self):
        """
        Hàm được gọi sau khi một người đánh bài.

        Khi xuất hiện người về nhất:
        - Người còn nguyên 13 lá bị lủng.
        - Người lủng ngừng chơi và nhận hạng chót.
        - Những người còn lại tiếp tục nếu cần.
        """
        if not hasattr(self, "lung_seats"):
            self.lung_seats = set()

        # finished chỉ chứa những người đã đánh hết bài hợp lệ.
        # Tới trắng được xử lý riêng bởi tl_check_white().
        if self.finished and not getattr(
            self, "lung_checked", False
        ):
            self.lung_checked = True
            first_winner = self.finished[0]

            for seat in range(self.n):
                if (
                    seat == first_winner
                    or seat in self.finished
                    or seat in self.thoi_two
                ):
                    continue

                # Trong Tiến lên không có bốc/ăn bài,
                # còn 13 lá nghĩa là chưa đánh được lá nào.
                if len(self.hands[seat]) == 13:
                    self.lung_seats.add(seat)
                    self.passed.discard(seat)

                    self.say(
                        f"💀 {self.name(seat)} LỦNG 13 LÁ "
                        f"khi {self.name(first_winner)} về nhất "
                        f"— nhận hạng {self.n}, ngừng đánh."
                    )

        active = self.tl_active()

        # Còn ít nhất hai người hợp lệ: tiếp tục tranh hạng.
        if len(active) > 1:
            return False

        ranks = {}

        # Những người đã hết bài giữ thứ tự về thực tế.
        for position, seat in enumerate(self.finished, 1):
            ranks[seat] = position

        # Người hợp lệ cuối cùng nhận vị trí kế tiếp.
        if active:
            ranks[active[0]] = len(self.finished) + 1

        # Lủng và thối đều đồng hạng thấp nhất.
        last_place_seats = (
            set(self.thoi_two) | self.lung_seats
        )

        for seat in last_place_seats:
            ranks[seat] = self.n

        if len(ranks) != self.n:
            raise RuntimeError(
                "Thiếu thứ hạng khi kết thúc Tiến lên."
            )

        # Hàm có sẵn lưu rank_map và kết thúc ván.
        self.tl_finish_ranks(ranks)
        return True

    def analyze(self):
        if self.kind != "Tiến lên":
            return old_analyze(self)

        lung = getattr(self, "lung_seats", set())

        if self.done:
            lines = [self.status, ""]

            for seat in sorted(
                range(self.n),
                key=lambda item: (
                    self.rank_map[item],
                    item,
                ),
            ):
                notes = []

                if seat in lung:
                    notes.append("LỦNG 13 LÁ")

                if seat in self.thoi_two:
                    notes.append("THỐI 2")

                if seat in self.white_labels:
                    notes.append(self.white_labels[seat])

                suffix = (
                    " — " + ", ".join(notes)
                    if notes else ""
                )

                lines.append(
                    f"Hạng {self.rank_map[seat]}: "
                    f"{self.name(seat)}{suffix}"
                )

            if lung:
                lines.extend([
                    "",
                    "Người lủng đồng hạng chót.",
                    "Tournament bàn 4 người: "
                    "mỗi người hạng 4 nhận 1 điểm.",
                ])

            return "\n".join(lines), []

        human = self.human_seat()

        if human is not None and human in lung:
            return (
                "Bạn đã LỦNG 13 LÁ.\n\n"
                "Khi có người về nhất, bạn chưa đánh "
                "được lá nào nên nhận hạng "
                f"{self.n}.\n\n"
                "Bạn không đánh tiếp trong ván này. "
                "Các ghế còn lại tiếp tục tranh hạng."
            ), []

        return old_analyze(self)

    def seat_details(self, seat):
        game = self.game

        if (
            game is not None
            and game.kind == "Tiến lên"
            and seat in getattr(game, "lung_seats", set())
        ):
            return f"LỦNG 13 LÁ • Hạng {game.n}"

        return old_seat_details(self, seat)

    Engine.start_tl = start_tl
    Engine.tl_active = tl_active
    Engine.tl_finish_if_needed = tl_finish_if_needed
    Engine.analyze = analyze

    Lounge.seat_details = seat_details

    namespace["_TIENLEN_LUNG_INSTALLED"] = True

def install_tournament_penalties(namespace):
    """Bổ sung cơ chế trừ điểm vào class Tournament."""
    Tournament = namespace["Tournament"]

    if getattr(Tournament, "_penalty_rules_installed", False):
        return

    def apply_card_penalty(
        self,
        *,
        hand_id,
        action_id,
        victim_pid,
        kind,
        two_count=0,
    ):
        """
        hand_id:
            ID duy nhất của ván trong giải.
            Nếu có nhiều bàn, ID phải gồm cả bàn.

        action_id:
            ID của hành động ăn/chặt đã được xác nhận.
            Dùng lại cùng ID nếu xử lý lại cùng hành động,
            để tránh trừ điểm hai lần.

        victim_pid:
            ID người bị ăn/chặt, KHÔNG phải số ghế.

        kind:
            "ta_la_eaten" hoặc "tien_len_chopped".

        two_count:
            Số con 2 trong bộ bài BỊ CHẶT.
            Nếu không phải bộ 2, truyền 0.

        Trả về:
            Số điểm đã trừ; 0 nếu sự kiện đã xử lý.
        """
        if kind == "ta_la_eaten":
            units = 1

        elif kind == "tien_len_chopped":
            if type(two_count) is not int or two_count < 0:
                raise ValueError("two_count phải là số nguyên >= 0.")

            # Mỗi con 2 = một đơn vị phạt 0.5 điểm.
            # Bộ khác bị chặt: một lần = 0.5 điểm.
            units = two_count if two_count > 0 else 1

        else:
            raise ValueError(f"Loại phạt không hợp lệ: {kind}")

        if not hasattr(self, "_card_penalty_seen"):
            self._card_penalty_seen = set()
            self.card_penalty_log = []

        event_key = (
            hand_id,
            action_id,
            kind,
            victim_pid,
        )

        if event_key in self._card_penalty_seen:
            return 0.0

        # Chỉ cập nhật người đã được đăng ký trong giải.
        if (
            victim_pid not in self.points
            or victim_pid not in self.total_points
        ):
            raise KeyError(
                f"Người chơi {victim_pid!r} chưa có điểm Tournament."
            )

        amount = units * 0.5

        # Không dùng max(0, ...): cho phép điểm âm.
        self.points[victim_pid] -= amount

        # Tránh trừ hai lần nếu hai tên cùng trỏ một bảng.
        if self.total_points is not self.points:
            self.total_points[victim_pid] -= amount

        self._card_penalty_seen.add(event_key)

        self.card_penalty_log.append({
            "hand_id": hand_id,
            "action_id": action_id,
            "victim_pid": victim_pid,
            "kind": kind,
            "two_count": two_count,
            "delta": -amount,
        })

        return amount

    Tournament.apply_card_penalty = apply_card_penalty
    Tournament._penalty_rules_installed = True


def format_tournament_score(value):
    """Hiển thị 4, 3.5, -0.5; không làm mất phần thập phân."""
    return f"{value:g}"
